\chapter{The problem and the result}
\label{ch:problem}

Let $(\Omega, \mu)$ be a measure space, $\iota$ a finite nonempty index set with
$m = \#\iota$ elements and $\kappa$ a second index set, finite or countably infinite.
Two families of
square-integrable functions on $\Omega$ are given,
\[
  a = (a_k)_{k \in \iota}, \qquad b = (b_p)_{p \in \kappa},
\]
and we ask whether the integral of $|f|^2$ can be replaced by finitely many point
evaluations. Let $J$ be the Gram matrix of the second family, let $\Lambda > 0$ satisfy
$J \preceq \Lambda \cdot 1$, and let $M = \Tr J / \Lambda$ be its \emph{effective
dimension}. Then for every $n \geq m$ there are points $x_1, \dots, x_n \in \Omega$ and
weights $w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \int_\Omega |f|^2 \, d\mu
    \; \leq \; \sum_{i=1}^n w_i \, |f(x_i)|^2
  \qquad \text{for all } f \in \operatorname{span}\{a_k\},
\]
and
\[
  \sum_{i=1}^n w_i \, |g(x_i)|^2
    \; \leq \; \Bigl(1 + \sqrt{\tfrac{M-1}{n}}\Bigr)^2 \Lambda \sum_p |c_p|^2
  \qquad \text{for } g = \sum_p c_p \, b_p .
\]
This is \Cref{thm:sparsification} and \Cref{cor:l2-discretization}. The two families
enter asymmetrically: the first through its size $m$, the second only through $M$, which
is at most $\#\kappa$ and much smaller as soon as the eigenvalues of $J$ decay. Taking
$a = b$ with both Gram matrices the identity gives $M = m$ and the sparsification theorem
of Batson, Spielman and Srivastava~\cite{BSS}, which is \Cref{cor:bss}.

The last chapter answers which measure to discretize in the first place. There $D$ is an
arbitrary set carrying $m$ bounded, linearly independent functions, and Kiefer and
Wolfowitz give, for every $\varepsilon > 0$, finitely many points and weights with
$|f(y)|^2 \leq (m + \varepsilon) \sum_k w_k |f(x_k)|^2$ for every $y \in D$ and every $f$
in their span: the uniform norm is dominated by the $L_2$ norm of that measure. Feeding
such a measure to \Cref{thm:sparsification} is how one reaches sampling projections.

A statement about these weighted sums is a statement about the matrices
$\sum_i w_i \, a(x_i) a(x_i)^*$ and $\sum_i w_i \, b(x_i) b(x_i)^*$, compared in the
Loewner order: $A \preceq B$ means that $B - A$ is positive semidefinite. That is the
form the argument takes. Traces appearing below are real, and the formalisation takes
real parts wherever a real number is needed; it writes a function of a span as
$f = \langle c, a(\cdot) \rangle$, which conjugates the coefficient vector, and since $c$
runs over all of $\mathbb{C}^{\iota}$ that is the same set of functions.

Each statement names the Lean declaration that formalises it. Every statement and every
proof is formalised: the project compiles without \texttt{sorry}, so each result is
verified by the Lean kernel.

For a family $a : \Omega \to \mathbb{C}^{\iota}$ that is square-integrable in every
coordinate, the \emph{Gram matrix} is
\[
  G(a) = \int_\Omega a(x) a(x)^* \, d\mu(x) \in \mathbb{C}^{\iota \times \iota},
\]
the matrix with entries $\int_\Omega a_k \overline{a_l} \, d\mu$; in Lean it is
\texttt{Discretization.gram}. The same formula defines the Gram matrix
$J = (\int_\Omega b_p \overline{b_q} \, d\mu)_{p, q \in \kappa}$ of a countable family
$b = (b_p)_{p \in \kappa}$; if $\sum_p \|b_p\|_{L_2}^2 < \infty$, it acts on
$\ell_2(\kappa)$ as a positive operator of trace $\sum_p \|b_p\|_{L_2}^2$. For $\Lambda > 0$
with $J \preceq \Lambda \cdot 1$, the number $M = \Tr J / \Lambda$ is the \emph{effective
dimension} of $J$ relative to $\Lambda$. The upper frame bound is stated in it, and it is
what makes the theorem applicable when the second family is infinite.

\begin{theorem}[Generalized sparsification]
\label{thm:sparsification}
\lean{Discretization.bss_generalized, Discretization.Family.bss_generalized,
      Discretization.bss_generalized_of_gram_eq_one',
      Discretization.bss_of_gram_eq_one, Discretization.bss_generalized_of_unique,
      Discretization.bss_generalized_of_small_dim,
      Discretization.bss_generalized_of_unique_of_small_dim}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let
$\kappa$ be a nonempty set that is finite or countably infinite. Let
$a : \Omega \to \mathbb{C}^{\iota}$ be square-integrable in every coordinate with
$I = \int_\Omega a(x) a(x)^* \, d\mu(x)$ positive definite, and let
$b = (b_p)_{p \in \kappa}$ be square-integrable functions on $\Omega$ with
$\sum_p \|b_p\|_{L_2}^2 < \infty$. Let $J$ be their Gram matrix, with entries
$J_{pq} = \int_\Omega b_p \overline{b_q} \, d\mu$, as an operator on $\ell_2(\kappa)$;
assume that $J$ is injective, let $\Lambda > 0$ satisfy $J \preceq \Lambda \cdot 1$, and
put $m = \#\iota$ and $M = \Tr J / \Lambda = \sum_p \|b_p\|_{L_2}^2 / \Lambda$. Then for
every $n \geq m$ there are points $x_1, \dots, x_n \in \Omega$ with
$\sum_p |b_p(x_i)|^2 < \infty$ and weights $w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot I
    \; \preceq \; \sum_{i=1}^n w_i \, a(x_i) a(x_i)^*
\]
and
\[
  \sum_{i=1}^n w_i \, \Bigl| \sum_{p \in \kappa} c_p \, b_p(x_i) \Bigr|^2
    \; \leq \; \Bigl(1 + \sqrt{\tfrac{M-1}{n}}\Bigr)^2 \Lambda \sum_{p \in \kappa} |c_p|^2
  \qquad \text{for every } c \in \ell_2(\kappa).
\]
\end{theorem}

\begin{proof}
\leanok
\uses{lem:gram-identities, thm:two-barriers, lem:barrier, lem:verifiers-on-average,
      lem:point-from-averages, lem:frame-bounds-from-potentials, lem:loewner-basic-bounds,
      lem:family-gram-operator, thm:operator-sparsification}
\emph{Finite $\kappa$.} Here $b(x) = (b_p(x))_p \in \mathbb{C}^{\kappa}$, and the upper
bound is the Loewner inequality
$\sum_i w_i \, b(x_i) b(x_i)^* \preceq (1 + \sqrt{(M-1)/n})^2 \Lambda \cdot 1$ read at the
vector $\bar c$. Replace $a$ by $I^{-1/2} a$, whose Gram matrix is $I^{-1/2} I \, I^{-1/2} = 1$ by
\Cref{lem:gram-identities}(b). Conjugating the resulting lower bound by $I^{1/2}$
preserves the Loewner order and turns $(1 - r)^2 \cdot 1$, with $r = \sqrt{(m-1)/n}$,
into $(1 - r)^2 \cdot I$; this is \texttt{Discretization.bss\_of\_gram\_eq\_one}, and the
upper bound does not involve the first family. It remains to prove the claim for $I = 1$,
which is \texttt{Discretization.bss\_generalized\_of\_gram\_eq\_one'}. For $m \geq 2$ and
$M \geq 1 + 1/n$ it is \Cref{thm:two-barriers}. Otherwise one of the two potentials is not
needed: the verifier on that side is replaced by a constant, whose average is again $n$.
For $m = 1$ the lower constant $(1 - r)^2$ is $1$.

Case $m = 1$, $M \geq 1 + 1/n$ (\texttt{Discretization.bss\_generalized\_of\_unique}). The
lower verifier is replaced by $L(x) = n |a(x)|^2$, whose average is $n$ because the single
function is normalized. The induction then tracks only the upper matrix, using the upper
half of \Cref{lem:barrier} and the upper average of \Cref{lem:verifiers-on-average} with
\Cref{lem:point-from-averages}(a), and takes the largest weight the constant allows,
$1/w_i = L(x_i)$. Each point therefore contributes exactly $1/n$ to the lower bound, which
becomes the identity $\sum_i w_i |a(x_i)|^2 = 1$. The upper bound is
\Cref{lem:frame-bounds-from-potentials}(b).

Case $m \geq 2$, $M \leq 1 + 1/n$
(\texttt{Discretization.bss\_generalized\_of\_small\_dim}). The upper verifier is replaced
by $U(x) = n \|b(x)\|^2 / \Tr J$, whose average is $n$. The induction tracks only the lower
matrix, with the lower half of \Cref{lem:barrier}, and the lower bound is
\Cref{lem:frame-bounds-from-potentials}(a). The weights it produces satisfy
$w_i U(x_i) \leq 1$, that is $w_i \|b(x_i)\|^2 \leq \Tr J / n$. Summing over the $n$
points and using \Cref{lem:loewner-basic-bounds}(c) gives
$\sum_i w_i \, b(x_i) b(x_i)^* \preceq \Tr J \cdot 1 = M \Lambda \cdot 1$, and
$M \leq 1 + 1/n$ forces $s = \sqrt{(M-1)/n} \leq 1/n$ and hence
$M = 1 + n s^2 \leq 1 + s \leq (1+s)^2$.

Case $m = 1$, $M \leq 1 + 1/n$
(\texttt{Discretization.bss\_generalized\_of\_unique\_of\_small\_dim}). Both verifiers are
now the constants $L(x) = n |a(x)|^2$ and $U(x) = n \|b(x)\|^2 / \Tr J$, of equal average
$n$, and neither depends on the state of the construction.
\Cref{lem:point-from-averages}(b) produces a single point $y$ with $U(y) \leq L(y)$ and
$L(y) > 0$; using it $n$ times with the weight $w = 1/L(y)$ makes the lower bound the
identity $\sum_i w_i |a(x_i)|^2 = 1$, while $w \, U(y) \leq 1$ and
\Cref{lem:loewner-basic-bounds}(c) give the upper bound as in the previous case. No
induction is needed.

\emph{Countably infinite $\kappa$.} By \Cref{lem:family-gram-operator}(a) there is a
measurable set $S \subseteq \Omega$ of full measure on which $b(x) = (b_p(x))_p$ lies in
$\ell_2(\kappa)$. Restricted to $S$, which changes neither $I$ nor any of the integrals,
$b$ is a square-integrable map into $\ell_2(\kappa)$ whose Gram operator is $J$, positive,
injective, bounded by $\Lambda \cdot 1$ and of trace $\sum_p \|b_p\|^2$ along the standard
basis, by \Cref{lem:family-gram-operator}(b). \Cref{thm:operator-sparsification}, applied
to the measure space $S$ and the space $\ell_2(\kappa)$, which is separable because $\kappa$
is countable, gives the points, which therefore lie in $S$, and the weights;
its upper bound in the order of operators, evaluated at $\bar c$, is the bound claimed.
\end{proof}

In the formalisation the Gram matrix of the second family enters through its quadratic
form $\langle c, J c \rangle = \int_\Omega |\sum_p \overline{c_p} \, b_p|^2 \, d\mu$:
$J \preceq \Lambda \cdot 1$ is the bound of this integral by $\Lambda \|c\|^2$ for every
$c \in \ell_2(\kappa)$, and injectivity says that it vanishes only for $c = 0$. For finite
$\kappa$ the theorem is \texttt{Discretization.bss\_generalized}, where injectivity is
positive definiteness and both finiteness conditions are automatic; for countable
$\kappa$ it is \texttt{Discretization.Family.bss\_generalized}.

This is Theorem 3 of Chkifa, Dolbeault, Krieg and Ullrich~\cite{CDKU}. In eigenvalue form
the first bound is the factor $(1 - \sqrt{(m-1)/n})^2 \lambda_{\min}(I)$ of the paper. No
side condition on $m$ or on $M$ is needed; the potential argument of
\Cref{ch:construction} does need two, and the proof above removes them.

\begin{corollary}[Sparsification]
\label{cor:bss}
\lean{Discretization.bss}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let
$a : \Omega \to \mathbb{C}^{\iota}$ be square-integrable in every coordinate with
$\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$. Put $m = \#\iota$. Then for every $n \geq m$ there are points
$x_1, \dots, x_n \in \Omega$ and weights $w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot 1
    \; \preceq \; \sum_{i=1}^n w_i \, a(x_i) a(x_i)^*
    \; \preceq \; \Bigl(1 + \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot 1 .
\]
\end{corollary}

\begin{proof}
\leanok
\uses{thm:sparsification}
This is \Cref{thm:sparsification} for $b = a$ and $I = 1$: the Gram matrix of the second
family is then the identity, $\Lambda = 1$ bounds it, and its effective dimension is
$\Tr 1 / 1 = m$, so both frame constants are built from the same $r = \sqrt{(m-1)/n}$.
\end{proof}

This is the theorem of Batson, Spielman and Srivastava~\cite{BSS} in a slightly stronger
form. They have $\sqrt{m/n}$ in place of $\sqrt{(m-1)/n}$, so for $n = d\,m$ the ratio of
the upper to the lower constant, the condition number, is
$\bigl((\sqrt{d}+1)/(\sqrt{d}-1)\bigr)^2$. The $m-1$ in place of $m$ comes
from~\cite{CDKU}. The counting measure on a finite set turns the corollary into their
statement for finitely many vectors.

Read through quadratic forms, the theorem becomes a statement about norms.

\begin{corollary}[Discretization of the $L_2$-norm]
\label{cor:l2-discretization}
\lean{Discretization.exists_discretization, Discretization.Family.exists_discretization,
      Discretization.Infinite.exists_discretization,
      Discretization.Infinite.exists_discretization_of_hilbertBasis,
      Discretization.mul_integral_norm_sq_le_sum,
      Discretization.Infinite.sum_mul_norm_sq_inner_le}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let
$\kappa$ be a nonempty set that is finite or countably infinite. Let
$a : \Omega \to \mathbb{C}^{\iota}$ be square-integrable in every coordinate with
$\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$, and let $b = (b_p)_{p \in \kappa}$ be
square-integrable functions on $\Omega$ with $\sum_p \|b_p\|_{L_2}^2 < \infty$ whose Gram
matrix $J = (\int_\Omega b_p \overline{b_q} \, d\mu)_{p, q}$, as an operator on
$\ell_2(\kappa)$, is injective. Let $\Lambda > 0$ satisfy $J \preceq \Lambda \cdot 1$, and
put $m = \#\iota$ and $M = \Tr J / \Lambda$. Then for every $n \geq m$ there are points
$x_1, \dots, x_n \in \Omega$ with $\sum_p |b_p(x_i)|^2 < \infty$ and weights
$w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \int_\Omega |f|^2 \, d\mu
  \; \leq \; \sum_{i=1}^n w_i \, |f(x_i)|^2
\]
for every $f \in \operatorname{span}\{a_k\}$, and with
\[
  \sum_{i=1}^n w_i \, |g(x_i)|^2
  \; \leq \; \Bigl(1 + \sqrt{\tfrac{M-1}{n}}\Bigr)^2 \Lambda \sum_{p} |c_p|^2
\]
for every $c \in \ell_2(\kappa)$ and $g = \sum_p c_p \, b_p$.
\end{corollary}

\begin{proof}
\leanok
\uses{thm:sparsification, lem:gram-identities}
A Loewner inequality between matrices is an inequality between quadratic forms, uniformly
in the coefficient vector. For $f = \sum_k c_k a_k$ one has
$\sum_i w_i |f(x_i)|^2 = c^* G c$ with $G$ the accumulated matrix, and
$\int_\Omega |f|^2 d\mu = c^* G(a) c = \|c\|^2$ by \Cref{lem:gram-identities}(a) and
$G(a) = 1$. The second bound is that of \Cref{thm:sparsification}, applied with $I = 1$;
the series $g(x_i) = \sum_p c_p \, b_p(x_i)$ converges because
$\sum_p |b_p(x_i)|^2 < \infty$.
\end{proof}

The lower bound is stated for a function and the upper one for a coefficient vector,
because its right-hand side is the coefficient norm and the second family is not assumed
linearly independent, so a function of its span need not determine the coefficients.
This is Corollary 4 of~\cite{CDKU}. If the second family is the singular basis of the
embedding of a reproducing kernel Hilbert space $H$ into $L_2(\mu)$, that is, an
orthonormal basis of $H$ which is orthogonal in $L_2$, then the right-hand side is the
squared $H$-norm and $\Lambda$ is the squared norm of the embedding.

\chapter{Matrices in the Loewner order}
\label{ch:loewner}

The construction keeps two matrices positive definite and controls them through two
real numbers. This chapter collects what is needed about the Loewner order, the trace
and rank-one updates. Nothing here mentions sampling; all of it is stated for a general
field $\mathbb{C}$ or $\mathbb{R}$ where the C$^*$-algebra structure is not needed, and
none of it is in Mathlib.

\section{Comparison with multiples of the identity}

\begin{lemma}[Comparison with multiples of the identity]
\label{lem:loewner-basic-bounds}
\lean{Matrix.IsHermitian.le_smul_one, Matrix.PosSemidef.le_trace_smul_one,
      Matrix.vecMulVec_le_norm_sq_smul_one, Matrix.PosDef.inv_le_inv_of_le,
      Matrix.PosDef.smul_one_le_of_inv_le}
\leanok
Let $\iota$ be a finite set; all matrices below are in $\mathbb{C}^{\iota \times \iota}$.

(a) Let $A$ be Hermitian and $c$ a real number with $\lambda_i(A) \leq c$ for every
eigenvalue. Then $A \preceq c \cdot 1$.

(b) Every positive semidefinite $A$ satisfies $A \preceq (\Tr A) \cdot 1$.

(c) Every vector $v \in \mathbb{C}^{\iota}$ satisfies $v v^* \preceq \|v\|^2 \cdot 1$.

(d) If $A$ and $B$ are positive definite with $A \preceq B$, then
$B^{-1} \preceq A^{-1}$.

(e) If $A$ is positive definite and $A^{-1} \preceq c \cdot 1$ for some $c > 0$, then
$c^{-1} \cdot 1 \preceq A$.
\end{lemma}

\begin{proof}
\leanok
(a) By the spectral theorem $c \cdot 1 - A$ is unitarily conjugate to the diagonal
matrix with entries $c - \lambda_i(A) \geq 0$, and conjugation by a unitary preserves
positive semidefiniteness.

(b) The trace is the sum of the eigenvalues of $A$, all of which are nonnegative, so
every single eigenvalue is at most $\Tr A$, and (a) applies with $c = \Tr A$.

(c) The trace of $v v^*$ is $\|v\|^2$, so this is (b).

(d), (e) Both are read off Mathlib's antitonicity of the inverse in a C$^*$-algebra with
a star ordering, applied to the units of $\mathbb{C}^{\iota \times \iota}$ with the
$\ell_2$-operator norm; the inverse of $c \cdot 1$ is $c^{-1} \cdot 1$.
\end{proof}

\begin{lemma}[Conjugation by the square root]
\label{lem:sqrt-conjugation}
\lean{CStarAlgebra.inv_smul_le_of_conj_inv_sqrt_le}
\leanok
Let $A$ be a C$^*$-algebra with a star ordering, let $B \in A$ be positive and
invertible, let $X \in A$ and let $t > 0$. If
$B^{-1/2} X B^{-1/2} \preceq t \cdot 1$, then $t^{-1} \cdot X \preceq B$.
\end{lemma}

\begin{proof}
\leanok
Conjugating the hypothesis by $B^{1/2}$ preserves the order. On the left the two
inverse square roots cancel, on the right $t \cdot 1$ becomes $t \cdot B$; dividing by
$t$ gives the claim.
\end{proof}

The next lemma is the reason the whole argument can be run on two real numbers: a bound
on the trace of an inverse is a bound on the matrix itself.

\begin{lemma}[Bounds from the trace of an inverse]
\label{lem:inverse-trace-bounds}
\lean{Matrix.PosDef.inv_re_trace_smul_one_le, Matrix.PosDef.inv_re_trace_mul_smul_le,
      Matrix.PosDef.sub_smul_one, Matrix.PosDef.add_smul}
\leanok
Let $\iota$ and $\kappa$ be finite sets.

(a) Every positive definite $A \in \mathbb{C}^{\iota \times \iota}$ satisfies
\[
  \bigl( \Tr A^{-1} \bigr)^{-1} \cdot 1 \; \preceq \; A .
\]

(b) All positive definite $B, J \in \mathbb{C}^{\kappa \times \kappa}$ satisfy
\[
  \bigl( \Tr (J B^{-1}) \bigr)^{-1} \cdot J \; \preceq \; B .
\]

(c) If $A \in \mathbb{C}^{\iota \times \iota}$ is positive definite and
$0 < \delta < (\Tr A^{-1})^{-1}$, then $A - \delta \cdot 1$ is positive definite.

(d) If $B, J \in \mathbb{C}^{\kappa \times \kappa}$ are positive definite and
$\zeta > 0$, then $B + \zeta \cdot J$ is positive definite.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:loewner-basic-bounds, lem:sqrt-conjugation, lem:trace-inequalities}
(a) \Cref{lem:loewner-basic-bounds}(b) applied to $A^{-1}$ gives
$A^{-1} \preceq (\Tr A^{-1}) \cdot 1$, and \Cref{lem:loewner-basic-bounds}(e) turns that
around.

(b) Write $S = B^{1/2}$. The matrix $S^{-1} J S^{-1}$ is positive semidefinite and has
trace $\Tr (J B^{-1})$, so \Cref{lem:loewner-basic-bounds}(b) bounds it by
$\Tr (J B^{-1}) \cdot 1$. That trace is positive by \Cref{lem:trace-inequalities}(a), so
\Cref{lem:sqrt-conjugation} applies and gives the claim.

(c) Write
$A - \delta \cdot 1 = \bigl( A - (\Tr A^{-1})^{-1} \cdot 1 \bigr)
  + \bigl( (\Tr A^{-1})^{-1} - \delta \bigr) \cdot 1$, a positive semidefinite matrix by
(a) plus a positive definite one.

(d) This is the sum of a positive definite matrix and a positive multiple of one.
\end{proof}

\section{Traces of products}

\begin{lemma}[Traces of products]
\label{lem:trace-inequalities}
\lean{Matrix.PosSemidef.trace_mul_nonneg, Matrix.PosDef.re_trace_mul_pos,
      Matrix.PosSemidef.norm_trace_mul_sq_le, Matrix.PosSemidef.re_trace_mul_sq_le}
\leanok
Let $\iota$ be a finite set; all matrices below are in $\mathbb{C}^{\iota \times \iota}$.

(a) For positive semidefinite $P$ and $Q$ we have $\Tr (P Q) \geq 0$. If $P$ is
positive definite, $Q$ is positive semidefinite and $Q \neq 0$, then
$\Tr (P Q) > 0$.

(b) Let $Y$ be positive semidefinite and $Z$ any matrix. Then
\[
  \bigl| \Tr (Z Y) \bigr|^2 \; \leq \; \Tr Y \cdot \Tr (Z Y Z^*) .
\]
\end{lemma}

\begin{proof}
\leanok
(a) Write $Q$ as a sum of rank-one matrices $v v^*$. Then
$\Tr (P v v^*) = v^* P v \geq 0$, with strict inequality for $v \neq 0$ when $P$ is
positive definite, and at least one of the $v$ is nonzero when $Q \neq 0$.

(b) A positive semidefinite $Y$ makes $\langle P, Q \rangle = \Tr (Q Y P^*)$ a
semi-inner product on matrices. Cauchy--Schwarz for the pair $1$, $Z$ is the
inequality. No commutation of $Y$ and $Z$ is used.
\end{proof}

\section{Rank-one updates}

\begin{lemma}[Rank-one updates: Sherman--Morrison and the determinant]
\label{lem:rank-one-updates}
\lean{Matrix.inv_add_smul_vecMulVec, Matrix.trace_inv_add_smul_vecMulVec,
      Matrix.PosDef.add_smul_vecMulVec, Matrix.PosDef.sub_smul_vecMulVec,
      Matrix.det_smul_add_smul_vecMulVec}
\leanok
Let $\iota$ be a finite set with $m = \#\iota$ elements and let
$v \in \mathbb{C}^{\iota}$.

(a) Let $A$ be Hermitian and invertible and $t$ a scalar with
$1 + t \, v^* A^{-1} v \neq 0$. Then $A + t \, v v^*$ is invertible,
\[
  \bigl( A + t \, v v^* \bigr)^{-1}
    = A^{-1} - \frac{t}{1 + t \, v^* A^{-1} v} \,
      \bigl( A^{-1} v \bigr) \bigl( A^{-1} v \bigr)^* ,
\]
and consequently
\[
  \Tr \bigl( A + t \, v v^* \bigr)^{-1}
    = \Tr A^{-1} - \frac{t}{1 + t \, v^* A^{-1} v} \, v^* A^{-2} v .
\]

(b) Let $A$ be positive definite and $w \geq 0$. Then $A + w \, v v^*$ is positive
definite. If moreover $w \, v^* A^{-1} v < 1$, then $A - w \, v v^*$ is positive
definite.

(c) Let $A$ be invertible, $\beta \neq 0$ a scalar and $\alpha$ a scalar. Then
\[
  \det(\beta A + \alpha \, v v^*)
    = \beta^{\,m-1} \det A \, \bigl( \beta + \alpha \, v^* A^{-1} v \bigr).
\]
\end{lemma}

\begin{proof}
\leanok
(a) Multiplying the right-hand side by $A + t \, v v^*$ collapses, by Hermitian symmetry
of $A^{-1}$, to $1$ plus a multiple of $v (A^{-1} v)^*$ whose scalar factor vanishes by
the choice of the coefficient. Taking traces and using
$\Tr \bigl( (A^{-1} v)(A^{-1} v)^* \bigr) = v^* A^{-2} v$ gives the trace identity.

(b) The first is a positive definite matrix plus a positive semidefinite one. For the
second, (a) exhibits the inverse of $A - w \, v v^*$ as a positive definite matrix plus a
positive semidefinite one, and a matrix whose inverse is positive definite is itself
positive definite.

(c) Factor $\beta A$ out and apply the matrix determinant lemma
$\det(A + v w^*) = \det A \, (1 + w^* A^{-1} v)$, which holds because the added matrix
has rank one. The dependence on $\alpha$ is affine; this identity is what replaces the
derivative of the determinant.
\end{proof}

\chapter{Averages of quadratic forms}
\label{ch:averages-of-forms}

The bridge between the analytic side of the problem and the matrix algebra is one
identity: the average of a quadratic form along a family is a trace against its Gram
matrix. Read from right to left it says that a trace is an average, which is how the
construction finds a point.

\begin{lemma}[Two identities for the Gram matrix]
\label{lem:gram-identities}
\lean{Discretization.gram, Discretization.integral_quadForm,
      Discretization.integral_re_quadForm, Discretization.gram_mulVec}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite set, let
$a : \Omega \to \mathbb{C}^{\iota}$ be square-integrable in every coordinate, and write
$G(a) = \int_\Omega a(x) a(x)^* \, d\mu(x)$.

(a) For every matrix $Q \in \mathbb{C}^{\iota \times \iota}$,
\[
  \int_\Omega a(x)^* Q \, a(x) \, d\mu(x) = \Tr \bigl( Q \, G(a) \bigr).
\]

(b) For every matrix $S \in \mathbb{C}^{\iota \times \iota}$,
\[
  G(S a) = S \, G(a) \, S^* .
\]
\end{lemma}

\begin{proof}
\leanok
(a) Expand both sides as a double sum over $\iota$ and exchange the finite sum with the
integral. The integrand is integrable by Cauchy--Schwarz for $L_2$-functions.

(b) Expand both sides entrywise and exchange the finite sums with the integral.
\end{proof}

\begin{lemma}[A good point from two averages]
\label{lem:point-from-averages}
\lean{Discretization.exists_lt_of_integral_lt, Discretization.exists_le_of_integral_le}
\leanok
Let $(\Omega, \mu)$ be a measure space and let $f, g : \Omega \to \mathbb{R}$ be
integrable.

(a) If $\int_\Omega g \, d\mu < \int_\Omega f \, d\mu$, then $g(x) < f(x)$ for at least
one $x \in \Omega$.

(b) If $f$ and $g$ are nonnegative with $\int_\Omega g \, d\mu \leq \int_\Omega f \, d\mu$
and $\int_\Omega f \, d\mu > 0$, then there is an $x \in \Omega$ with $g(x) \leq f(x)$
and $f(x) > 0$.
\end{lemma}

\begin{proof}
\leanok
(a) If $f \leq g$ everywhere then $\int f \leq \int g$.

(b) If $f \leq g$ everywhere then the two averages agree, so the nonnegative function
$g - f$ has average zero and vanishes almost everywhere. At every point where $f$ and
$g$ agree, the assumption forces $f$ to vanish, so $f$ vanishes almost everywhere,
contradicting $\int f > 0$.
\end{proof}

The paper states step (a) as: the set of admissible points has positive measure. One
point is all the construction uses. When the two averages are equal, which happens in
the doubly degenerate case in the proof of \Cref{thm:sparsification}, a strict inequality
is no longer available
and (b) takes over. Both the nonnegativity of $g$ and the conclusion $f(x) > 0$ in (b)
are needed: for $f = (1, 0)$ and $g = (2, -1)$ on two points with the counting measure
both averages are $1$, and the only point with $g \leq f$ has $f = 0$.

\chapter{Potentials, verifiers and the barrier lemma}
\label{ch:barrier}

Two matrices are carried through the construction: a small one $A$, which must stay
above a multiple of the identity, and a large one $B$, which must stay below one. Each
is controlled by a single real number, and whether a candidate point may be used is
decided by two functions of that point.

\begin{definition}[Potentials and verifiers]
\label{def:potentials-and-verifiers}
\lean{Discretization.lowerPotential, Discretization.upperPotential,
      Discretization.lowerVerifier, Discretization.upperVerifier}
\leanok
Let $\iota$ and $\kappa$ be finite sets, let $A \in \mathbb{C}^{\iota \times \iota}$ and
$B, J \in \mathbb{C}^{\kappa \times \kappa}$ be positive definite, and let
$\delta, \zeta > 0$ with $A - \delta \cdot 1$ positive definite. The \emph{lower} and the
\emph{upper potential} are
\[
  \Phi(A) = \Tr A^{-1}, \qquad \Psi_J(B) = \Tr \bigl( J B^{-1} \bigr).
\]
With $Z = (A - \delta \cdot 1)^{-1}$ and $X = (B + \zeta \cdot J)^{-1}$, the \emph{lower}
and the \emph{upper verifier} are, for $v \in \mathbb{C}^{\iota}$ and
$u \in \mathbb{C}^{\kappa}$,
\[
  L_A^{\delta}(v)
    = \frac{v^* Z^2 v}{\Phi(A - \delta \cdot 1) - \Phi(A)} - v^* Z v,
  \qquad
  U_{B}^{\zeta}(u)
    = \frac{u^* X J X u}{\Psi_J(B) - \Psi_J(B + \zeta \cdot J)} + u^* X u .
\]
\end{definition}

The lower potential is large when $A$ has a small eigenvalue, the upper one when $B$ is
small in the directions where $J$ is large. Before each step $A$ is shrunk by
$\delta \cdot 1$ and $B$ is grown by $\zeta \cdot J$; both shifts move the potentials in
the unfavourable direction by an amount that is computed exactly, and that amount is the
budget one new sampling point is allowed to consume. The verifiers decide pointwise,
without recomputing any inverse, whether a candidate point stays within that budget.

\begin{lemma}[How the potentials change]
\label{lem:potential-change}
\lean{Discretization.lowerPotential_sub_eq, Discretization.upperPotential_sub_eq,
      Discretization.lowerPotential_add_smul_vecMulVec,
      Discretization.upperPotential_sub_smul_vecMulVec}
\leanok
\uses{def:potentials-and-verifiers}
Let $\iota$ and $\kappa$ be finite sets and let $J \in \mathbb{C}^{\kappa \times \kappa}$
be positive definite.

(a) Let $A \in \mathbb{C}^{\iota \times \iota}$ and
$B \in \mathbb{C}^{\kappa \times \kappa}$ be positive definite, let $\delta > 0$ with
$A - \delta \cdot 1$ positive definite and let $\zeta \geq 0$. Then
\[
  \Phi(A - \delta \cdot 1) - \Phi(A)
    = \delta \, \Tr \bigl( (A - \delta \cdot 1)^{-1} A^{-1} \bigr)
\]
and
\[
  \Psi_J(B) - \Psi_J(B + \zeta \cdot J)
    = \zeta \, \Tr \bigl( J B^{-1} J (B + \zeta \cdot J)^{-1} \bigr).
\]
Both right-hand sides are positive for $\delta, \zeta > 0$.

(b) Let $N \in \mathbb{C}^{\iota \times \iota}$ be positive definite,
$v \in \mathbb{C}^{\iota}$ and $w \geq 0$. Then
\[
  \Phi \bigl( N + w \, v v^* \bigr)
    = \Phi(N) - \frac{w}{1 + w \, v^* N^{-1} v} \, v^* N^{-2} v .
\]
Let $C \in \mathbb{C}^{\kappa \times \kappa}$ be positive definite,
$u \in \mathbb{C}^{\kappa}$ and $w \geq 0$ with $w \, u^* C^{-1} u < 1$. Then
\[
  \Psi_J \bigl( C - w \, u u^* \bigr)
    = \Psi_J(C) + \frac{w}{1 - w \, u^* C^{-1} u} \, u^* C^{-1} J C^{-1} u .
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:trace-inequalities, lem:inverse-trace-bounds, lem:rank-one-updates}
(a) Both identities follow from the resolvent identity
$X^{-1} - Y^{-1} = X^{-1} (Y - X) Y^{-1}$ with the difference of the two matrices being
$\delta \cdot 1$ and $\zeta \cdot J$ respectively; $B + \zeta \cdot J$ is positive
definite by \Cref{lem:inverse-trace-bounds}(d). The signs are
\Cref{lem:trace-inequalities}(a) applied to the products of positive definite matrices on
the right.

(b) Insert the Sherman--Morrison formula of \Cref{lem:rank-one-updates}(a) and take
traces.
\end{proof}

\begin{lemma}[Barrier lemma]
\label{lem:barrier}
\lean{Discretization.lowerPotential_update_le, Discretization.upperPotential_update_le}
\leanok
\uses{def:potentials-and-verifiers}
Let $\iota$ and $\kappa$ be finite sets. Let $A \in \mathbb{C}^{\iota \times \iota}$ be
positive definite, $0 < \delta < \Phi(A)^{-1}$, $v \in \mathbb{C}^{\iota}$, and let
$w > 0$ satisfy $1/w \leq L_A^{\delta}(v)$. Then $A - \delta \cdot 1 + w \, v v^*$ is
positive definite and
\[
  \Phi \bigl( A - \delta \cdot 1 + w \, v v^* \bigr) \leq \Phi(A).
\]
Let $B, J \in \mathbb{C}^{\kappa \times \kappa}$ be positive definite, $\zeta > 0$,
$u \in \mathbb{C}^{\kappa}$, and let $w > 0$ satisfy $U_B^{\zeta}(u) \leq 1/w$. Then
$B + \zeta \cdot J - w \, u u^*$ is positive definite and
\[
  \Psi_J \bigl( B + \zeta \cdot J - w \, u u^* \bigr) \leq \Psi_J(B).
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:potential-change, lem:rank-one-updates, lem:inverse-trace-bounds}
Both halves are the closed form of \Cref{lem:potential-change}(b) combined with one
inequality between real numbers, and positive definiteness is
\Cref{lem:rank-one-updates}(b), with \Cref{lem:inverse-trace-bounds}(c) and (d) for the
shifted matrices. In the upper half the hypothesis on $w$ first has to be turned into
positivity of the Sherman--Morrison denominator $1 - w \, u^* X u$, which is where
positive definiteness of $J$ enters.
\end{proof}

\chapter{The verifiers pass on average}
\label{ch:averages}

A point may be used if the upper verifier stays below the lower one. Such a point exists
because the inequality holds on average.

\begin{lemma}[The verifiers pass on average]
\label{lem:verifiers-on-average}
\lean{Discretization.integral_lowerVerifier_gt, Discretization.integral_upperVerifier_lt,
      Discretization.exists_admissible_point}
\leanok
\uses{def:potentials-and-verifiers}
Let $(\Omega, \mu)$ be a measure space, let $\iota$ and $\kappa$ be finite nonempty sets,
and let $a : \Omega \to \mathbb{C}^{\iota}$ and $b : \Omega \to \mathbb{C}^{\kappa}$ be
square-integrable in every coordinate with $\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$ and
$J = \int_\Omega b(x) b(x)^* \, d\mu(x)$ positive definite.
Let $A \in \mathbb{C}^{\iota \times \iota}$ and $B \in \mathbb{C}^{\kappa \times \kappa}$
be positive definite, and let $0 < \delta < \Phi(A)^{-1}$ and $\zeta > 0$. Then
\[
  \int_\Omega L_A^{\delta}\bigl( a(x) \bigr) d\mu(x) > \frac{1}{\delta} - \Phi(A)
  \qquad\text{and}\qquad
  \int_\Omega U_B^{\zeta}\bigl( b(x) \bigr) d\mu(x) < \frac{1}{\zeta} + \Psi_J(B).
\]
If moreover $1/\zeta + \Psi_J(B) \leq 1/\delta - \Phi(A)$, then there is an
$x \in \Omega$ with $U_B^{\zeta}\bigl( b(x) \bigr) < L_A^{\delta}\bigl( a(x) \bigr)$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:gram-identities, lem:trace-inequalities, lem:loewner-basic-bounds,
      lem:potential-change, lem:point-from-averages}
For the lower verifier write $Y = A^{-1}$ and $Z = (A - \delta \cdot 1)^{-1}$.
\Cref{lem:gram-identities}(a) turns the average into
$\Tr (Z^2) / (\delta \Tr (Y Z)) - \Tr Z$. Writing $t = \Tr (Y Z)$ and
$u = \Tr (Y Z^2)$, the claim reduces to $\delta t^2 < u$, which is what
\Cref{lem:trace-inequalities}(b) gives together with $\delta \Phi(A) < 1$.

For the upper verifier write $W = B^{-1}$ and $X = (B + \zeta \cdot J)^{-1}$ and let
$E = \Psi_J(B) - \Psi_J(B + \zeta \cdot J)$ be the gain of
\Cref{lem:potential-change}(a). By \Cref{lem:gram-identities}(a) the average equals
$\Tr (J X J X) / E + \Psi_J(B + \zeta \cdot J)$. Since $X \preceq W$ by
\Cref{lem:loewner-basic-bounds}(d), the numerator is at most $\Tr (J W J X) = E / \zeta$
by \Cref{lem:trace-inequalities}(a), so the first summand is at most $1/\zeta$, and the
second is strictly smaller than $\Psi_J(B)$.

Under the gap condition the average of the lower verifier exceeds $1/\delta - \Phi(A)$,
which is at least $1/\zeta + \Psi_J(B)$, which exceeds the average of the upper
verifier. Now apply \Cref{lem:point-from-averages}(a).
\end{proof}

Any weight $w$ between the two reciprocals is then admissible for both halves of
\Cref{lem:barrier}. The construction always takes the largest one, $w = 1/L_A^{\delta}$.

\chapter{The construction and the frame bounds}
\label{ch:construction}

\begin{proposition}[The barrier iteration]
\label{prop:barrier-iteration}
\lean{Discretization.lowerState, Discretization.upperState,
      Discretization.exists_points_weights}
\leanok
\uses{def:potentials-and-verifiers}
Let $(\Omega, \mu)$ be a measure space, let $\iota$ and $\kappa$ be finite nonempty sets,
and let $a : \Omega \to \mathbb{C}^{\iota}$ and $b : \Omega \to \mathbb{C}^{\kappa}$ be
square-integrable in every coordinate with $\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$ and
$J = \int_\Omega b(x) b(x)^* \, d\mu(x)$ positive definite.
Let $A_0 \in \mathbb{C}^{\iota \times \iota}$ and $B_0 \in \mathbb{C}^{\kappa \times \kappa}$
be positive definite and let $\delta, \zeta > 0$ satisfy the initial gap condition
\[
  \frac{1}{\zeta} + \Psi_J(B_0) \; \leq \; \frac{1}{\delta} - \Phi(A_0).
\]
For points $x_1, \dots, x_k \in \Omega$ and weights $w_1, \dots, w_k$ write
\[
  A_k = A_0 - k \delta \cdot 1 + \sum_{i=1}^k w_i \, a(x_i) a(x_i)^*,
  \qquad
  B_k = B_0 + k \zeta \cdot J - \sum_{i=1}^k w_i \, b(x_i) b(x_i)^* .
\]
Then for every $k \geq 0$ there are points $x_1, \dots, x_k \in \Omega$ and weights
$w_1, \dots, w_k > 0$ such that $A_k$ and $B_k$ are positive definite,
$\Phi(A_k) \leq \Phi(A_0)$ and $\Psi_J(B_k) \leq \Psi_J(B_0)$.
\end{proposition}

\begin{proof}
\leanok
\uses{lem:verifiers-on-average, lem:barrier}
Induction on $k$. The invariant keeps the gap open, so \Cref{lem:verifiers-on-average}
produces a point $x_{k+1}$; the weight is the reciprocal of its lower verifier value,
and \Cref{lem:barrier} restores the invariant.
\end{proof}

The number of steps in \Cref{prop:barrier-iteration} is arbitrary; only the initial gap
condition drives it. What ties the number of points to the frame bounds is the choice of
the initial matrices and of the two shifts, made in the proof of
\Cref{thm:two-barriers}. The next lemma turns the final potentials into frame bounds.

\begin{lemma}[From the final potentials to the frame bounds]
\label{lem:frame-bounds-from-potentials}
\lean{Discretization.lower_bound_of_state, Discretization.upper_bound_of_state,
      Discretization.lower_frame_bound, Discretization.upper_frame_bound}
\leanok
\uses{def:potentials-and-verifiers}
Let $\iota$ and $\kappa$ be finite nonempty sets, let $n \geq 1$, and let
$v_1, \dots, v_n \in \mathbb{C}^{\iota}$, $u_1, \dots, u_n \in \mathbb{C}^{\kappa}$ and
$w_1, \dots, w_n \in \mathbb{R}$.

(a) Let $r > 0$ with $r \neq 1$, put $\delta = (1 - r)/n$ and
$c_0 = \delta (n r^2 + 1)/r$, and let
$A_n = c_0 \cdot 1 - n \delta \cdot 1 + \sum_{i=1}^n w_i \, v_i v_i^*$. If $A_n$ is
positive definite and $\Phi(A_n) \leq r / \delta$, then
\[
  (1 - r)^2 \cdot 1 \; \preceq \; \sum_{i=1}^n w_i \, v_i v_i^* .
\]

(b) Let $J \in \mathbb{C}^{\kappa \times \kappa}$ be positive definite and $\Lambda > 0$
with $J \preceq \Lambda \cdot 1$, let $s \geq 1/n$ with
$\Tr J = \Lambda (n s^2 + 1)$, put $\zeta = (1 + s)/n$ and $d_0 = \zeta \Tr J / s$, and
let $B_n = d_0 \cdot 1 + n \zeta \cdot J - \sum_{i=1}^n w_i \, u_i u_i^*$. If $B_n$ is
positive definite and $\Psi_J(B_n) \leq s / \zeta$, then
\[
  \sum_{i=1}^n w_i \, u_i u_i^* \; \preceq \; (1 + s)^2 \Lambda \cdot 1 .
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:inverse-trace-bounds}
By \Cref{lem:inverse-trace-bounds}(a), $\Phi(A_n) \leq r/\delta$ gives
$(\delta / r) \cdot 1 \preceq A_n$; inserting the definition of $A_n$ gives
$(\delta/r + n\delta - c_0) \cdot 1 \preceq \sum_i w_i \, v_i v_i^*$, and
$\delta/r + n\delta - c_0 = n \delta (1 - r) = (1 - r)^2$. By
\Cref{lem:inverse-trace-bounds}(b), $\Psi_J(B_n) \leq s/\zeta$ gives
$(\zeta / s) \cdot J \preceq B_n$, hence
$\sum_i w_i \, u_i u_i^* \preceq d_0 \cdot 1 + (n\zeta - \zeta/s) \cdot J$. The
coefficient $n\zeta - \zeta/s = \zeta (n - 1/s)$ is nonnegative exactly because
$s \geq 1/n$, so $J \preceq \Lambda \cdot 1$ may be applied to it, and
$d_0 + (n\zeta - \zeta/s)\Lambda = \zeta \Lambda (n s + n) = (1 + s)^2 \Lambda$.
The two read-offs are \texttt{Discretization.lower\_bound\_of\_state} and
\texttt{Discretization.upper\_bound\_of\_state}; the arithmetic is in
\texttt{Discretization/Parameters.lean}.
\end{proof}

For $r = \sqrt{(m-1)/n}$ and $s = \sqrt{(M-1)/n}$ one has $n r^2 + 1 = m$ and
$n s^2 + 1 = M$, and $s \geq 1/n$ is the condition $M \geq 1 + 1/n$. The lemma is stated
in $r$ and $s$ because its proof never unfolds the square roots.

\begin{theorem}[Sparsification with both barriers]
\label{thm:two-barriers}
\lean{Discretization.bss_generalized_of_gram_eq_one}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ and $\kappa$ be finite nonempty sets,
and let $a : \Omega \to \mathbb{C}^{\iota}$ and $b : \Omega \to \mathbb{C}^{\kappa}$ be
square-integrable in every coordinate with $\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$ and
$J = \int_\Omega b(x) b(x)^* \, d\mu(x)$ positive definite. Let $\Lambda > 0$ satisfy
$J \preceq \Lambda \cdot 1$, and put $m = \#\iota$ and $M = \Tr J / \Lambda$. Let
$n \geq m$, and assume $m \geq 2$ and $M \geq 1 + 1/n$.
Then there are points $x_1, \dots, x_n \in \Omega$ and weights $w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot 1
    \; \preceq \; \sum_{i=1}^n w_i \, a(x_i) a(x_i)^*
\]
and
\[
  \sum_{i=1}^n w_i \, b(x_i) b(x_i)^*
    \; \preceq \; \Bigl(1 + \sqrt{\tfrac{M-1}{n}}\Bigr)^2 \Lambda \cdot 1 .
\]
\end{theorem}

\begin{proof}
\leanok
\uses{prop:barrier-iteration, lem:frame-bounds-from-potentials}
Put
\[
  r = \sqrt{\tfrac{m-1}{n}}, \quad s = \sqrt{\tfrac{M-1}{n}}, \quad
  \delta = \tfrac{1-r}{n}, \quad \zeta = \tfrac{1+s}{n}, \quad
  A_0 = \tfrac{\delta m}{r} \cdot 1, \quad B_0 = \tfrac{\zeta \Tr J}{s} \cdot 1 .
\]
Since $2 \leq m \leq n$ we have $0 < r < 1$, and $M \geq 1 + 1/n$ gives $s \geq 1/n$.
Then $\Phi(A_0) = r/\delta$ and $\Psi_J(B_0) = s/\zeta$, so the initial gap closes
exactly, $1/\delta - \Phi(A_0) = n = 1/\zeta + \Psi_J(B_0)$.
\Cref{prop:barrier-iteration} with $k = n$ gives points and weights with
$\Phi(A_n) \leq r/\delta$ and $\Psi_J(B_n) \leq s/\zeta$, and
\Cref{lem:frame-bounds-from-potentials} turns them into the two bounds, since
$n r^2 + 1 = m$ and $n s^2 + 1 = M$.
\end{proof}

\chapter{A countably infinite second family}
\label{ch:infinite}

Nothing in \Cref{thm:sparsification} ties the number of points to the size of the
second family: it is governed by the effective dimension alone, so the second family may
be infinite, and this chapter supplies that case. The argument runs for a single
square-integrable map $b : \Omega \to H$ into a Hilbert space with a countable Hilbert
basis; a family $(b_p)_{p \in \kappa}$ is the case $H = \ell_2(\kappa)$ with
$b(x) = (b_p(x))_p$ (\Cref{lem:family-gram-operator}). The Gram matrix becomes the
\emph{Gram operator} $J$ of $b$, the bounded operator on $H$ with
\[
  \langle u, J u \rangle = \int_\Omega \bigl| \langle u, b(x) \rangle \bigr|^2 d\mu(x)
  \qquad \text{for every } u \in H .
\]
The first family stays finite, so the lower half of the argument is unchanged, and only
the upper half has to be redone. Mathlib has no trace-class theory, so the trace is
built along a Hilbert basis.

The Gram operator is positive, since the right-hand side is nonnegative. Its trace along
any Hilbert basis $e$ with countable index is finite and equal to
$\int_\Omega \|b\|^2 \, d\mu$, by Parseval and monotone convergence
(\Cref{lem:countable-averages}(a)). The only genuine assumption on $J$ is therefore that
it is injective, which is what positive definiteness of the Gram matrix says for a finite
family, and the main theorem of this chapter assumes nothing else: no basis appears in its
statement. The lemmas leading to it work along a fixed Hilbert basis with countable
index, which a separable space always has.

\section{The trace of an operator along a basis}

\begin{definition}[Trace along a Hilbert basis]
\label{def:trace-along-basis}
\lean{ContinuousLinearMap.traceAlong, Discretization.Infinite.IsFiniteTracePos}
\leanok
Let $H$ be a complex Hilbert space and $e = (e_k)_{k \in \kappa}$ a Hilbert basis of $H$.
For a bounded operator $T$ on $H$ put
\[
  \Tr_e T = \sum_{k \in \kappa} \re \langle e_k, T e_k \rangle,
\]
with the convention that the sum is zero when the family is not summable. The operator
$T$ is \emph{of finite trace along $e$} if the family
$\bigl( \re \langle e_k, T e_k \rangle \bigr)_k$ is summable.
\end{definition}

In Lean, \texttt{IsFiniteTracePos e J} bundles three properties of $J$: positive, of
finite trace along $e$, and injective. For a positive operator neither the finiteness nor
the value of $\Tr_e T$ depends on the basis (\Cref{lem:operator-traces}(c)), so the basis
only records how the trace is computed. For a positive $T$ the terms are nonnegative,
because
$\re \langle x, T x \rangle = \|T^{1/2} x\|^2$, so $\Tr_e T$ is the squared
Hilbert--Schmidt norm of $T^{1/2}$.

\begin{lemma}[Parseval and Hilbert--Schmidt sums under adjoints]
\label{lem:hilbert-schmidt-adjoint}
\lean{HilbertBasis.hasSum_norm_sq_inner, ContinuousLinearMap.tsum_norm_sq_adjoint,
      ContinuousLinearMap.summable_norm_sq_adjoint_iff,
      ContinuousLinearMap.tsum_ofReal_norm_sq_adjoint_of_basis,
      ContinuousLinearMap.tsum_ofReal_norm_sq_of_isSelfAdjoint}
\leanok
Let $H$ be a complex Hilbert space with a Hilbert basis $e = (e_k)_{k \in \kappa}$.

(a) For every $x \in H$ the family $\bigl( |\langle e_k, x \rangle|^2 \bigr)_k$ has sum
$\|x\|^2$.

(b) For every bounded operator $T$ on $H$,
\[
  \sum_k \| T e_k \|^2 = \sum_k \| T^* e_k \|^2 ,
\]
and the family on the left is summable if and only if the one on the right is.

(c) For every bounded operator $T$ on $H$ and every second Hilbert basis
$e' = (e'_j)_{j \in \kappa'}$ of $H$,
\[
  \sum_k \| T e_k \|^2 = \sum_j \| T^* e'_j \|^2 \qquad \text{in } [0, \infty] .
\]
\end{lemma}

\begin{proof}
\leanok
(a) Take real parts in Mathlib's Parseval identity
$\sum_k \langle x, e_k \rangle \langle e_k, x \rangle = \langle x, x \rangle$.

(b) Expanding both sides by (a) gives the same double series
$\sum_{j,k} |\langle e_j, T e_k \rangle|^2$ with the two indices exchanged. The
rearrangement is carried out in $[0, \infty]$, where every sum is defined and the
interchange needs no hypothesis; summability is then read off the value being finite.

(c) The same computation, expanding $\|T e_k\|^2$ by (a) along $e'$ and
$\|T^* e'_j\|^2$ along $e$. For a self-adjoint $T$ it says that the Hilbert--Schmidt sum
does not depend on the basis.
\end{proof}

Part (b) is the hinge of the whole trace layer: it is what supplies the summability of
$P^{1/2} Q$, whose adjoint $Q P^{1/2}$ is visibly Hilbert--Schmidt.

\begin{lemma}[Traces of positive operators]
\label{lem:operator-traces}
\lean{ContinuousLinearMap.le_traceAlong_smul_one, ContinuousLinearMap.traceAlong_mul,
      ContinuousLinearMap.traceAlong_mul_pos,
      ContinuousLinearMap.summable_re_inner_iff_of_basis,
      ContinuousLinearMap.traceAlong_eq_of_basis}
\leanok
\uses{def:trace-along-basis}
Let $H$ be a complex Hilbert space with a Hilbert basis $e$.

(a) Let $T$ be a positive operator on $H$ of finite trace along $e$. Then
$T \preceq \Tr_e T \cdot 1$.

(b) Let $P$ be positive and of finite trace along $e$, and let $Q$ be positive. Then
\[
  \Tr_e (P Q) = \sum_k \re \bigl\langle P^{1/2} e_k, \, Q \, P^{1/2} e_k \bigr\rangle
  \; \geq \; 0 ,
\]
and the value is positive as soon as $Q$ is invertible and $P \neq 0$.

(c) Let $T$ be positive and let $e'$ be a second Hilbert basis of $H$. Then $T$ is of
finite trace along $e$ if and only if it is of finite trace along $e'$, and
$\Tr_e T = \Tr_{e'} T$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:hilbert-schmidt-adjoint}
(a) For $x \in H$, writing $x$ in the basis and applying the Cauchy--Schwarz inequality
to each term gives $\re \langle x, T x \rangle \leq \Tr_e T \cdot \|x\|^2$. This is the
operator analogue of \Cref{lem:loewner-basic-bounds}(b), and unlike the matrix proof it
uses no diagonalisation.

(b) The identity is the cyclicity $\Tr (PQ) = \Tr (P^{1/2} Q P^{1/2})$, read as a
symmetry of the pairing $\sum_k \langle S e_k, T e_k \rangle$ under passing to adjoints;
the required summability is \Cref{lem:hilbert-schmidt-adjoint}(b). Every term on the
right is nonnegative. For the strict inequality, invertibility of $Q$ makes $Q^{1/2}$
injective, so a vanishing trace would force $P^{1/2}$ to annihilate the whole basis.

(c) Along either basis the trace is $\sum \|T^{1/2} e_k\|^2$, and
\Cref{lem:hilbert-schmidt-adjoint}(c) for the self-adjoint operator $T^{1/2}$ gives the
same value in $[0, \infty]$ for both. If it is infinite, both traces are $0$ by the
convention of \Cref{def:trace-along-basis}; no summability has to be assumed.
\end{proof}

Invertibility of $Q$ here is what replaces positive definiteness. A positive operator of
finite trace on an infinite-dimensional space is compact, hence never bounded away from
zero, so positive definiteness is unavailable and the hypotheses on the two operators
have to be chosen differently: injectivity for $J$, and invertibility for the operator
the construction moves.

\begin{lemma}[Rank-one updates of operators]
\label{lem:operator-rank-one-updates}
\lean{ContinuousLinearMap.inverse_add_smul_rankOne,
      ContinuousLinearMap.rankOne_le_norm_sq_smul_one}
\leanok
Let $H$ be a complex Hilbert space and $u \in H$, and write $u u^*$ for the rank-one
operator $z \mapsto \langle u, z \rangle u$.

(a) Let $A$ be a self-adjoint invertible operator on $H$ and $t$ a scalar with
$1 + t \langle u, A^{-1} u \rangle \neq 0$. Then $A + t \, u u^*$ is invertible and
\[
  \bigl( A + t \, u u^* \bigr)^{-1}
    = A^{-1} - \frac{t}{1 + t \langle u, A^{-1} u \rangle} \,
      \bigl( A^{-1} u \bigr) \bigl( A^{-1} u \bigr)^* .
\]

(b) $u u^* \preceq \|u\|^2 \cdot 1$.
\end{lemma}

\begin{proof}
\leanok
(a) The candidate is verified to be a two-sided inverse, exactly as in
\Cref{lem:rank-one-updates}(a). In a noncommutative ring one product does not suffice,
but two do: they exhibit $A + t \, u u^*$ as a unit.

(b) A self-adjoint element $a$ of a C$^*$-algebra satisfies $a \preceq \|a\| \cdot 1$,
and $u u^*$ is self-adjoint with $\|u u^*\| = \|u\|^2$.
\end{proof}

\section{The upper potential of an operator}

For the operator the construction moves, the right notion is positivity together with
invertibility; it is Mathlib's \texttt{IsStrictlyPositive}, and the inverse is
\texttt{Ring.inverse}, defined on all operators and equal to the inverse on the
invertible ones.

\begin{definition}[Upper potential and verifier of an operator]
\label{def:operator-upper-potential}
\lean{Discretization.Infinite.upperPotential, Discretization.Infinite.upperVerifier}
\leanok
\uses{def:trace-along-basis}
Let $H$ be a complex Hilbert space with a Hilbert basis $e$, let $J$ be a positive
injective operator on $H$ of finite trace along $e$, let $B$ be a strictly positive
operator on $H$, that is positive and invertible, and let $\zeta > 0$. The \emph{upper
potential} of $B$ is $\Psi_J(B) = \Tr_e \bigl( J B^{-1} \bigr)$, and with
$X = (B + \zeta \cdot J)^{-1}$ the \emph{upper verifier} at $u \in H$ is
\[
  U_B^{\zeta}(u)
    = \frac{\re \langle u, X J X u \rangle}
           {\Psi_J(B) - \Psi_J(B + \zeta \cdot J)} + \re \langle u, X u \rangle .
\]
\end{definition}

\begin{lemma}[Barrier lemma for operators]
\label{lem:operator-barrier}
\lean{Discretization.Infinite.upperPotential_sub_eq,
      Discretization.Infinite.upperPotential_add_smul_lt,
      Discretization.Infinite.upperPotential_update_le,
      Discretization.Infinite.inv_upperPotential_smul_le,
      Discretization.Infinite.upper_bound_of_state,
      Discretization.Infinite.upper_frame_bound}
\leanok
\uses{def:operator-upper-potential}
Let $H$ be a nonzero complex Hilbert space with a Hilbert basis $e$, let $J$ be a
positive injective operator on $H$ of finite trace along $e$, and let $B$ be a strictly
positive operator on $H$.

(a) For every $\zeta \geq 0$,
\[
  \Psi_J(B) - \Psi_J(B + \zeta \cdot J)
    = \zeta \, \Tr_e \bigl( (J B^{-1} J) (B + \zeta \cdot J)^{-1} \bigr),
\]
and for $\zeta > 0$ the right-hand side is positive.

(b) Let $\zeta > 0$, $u \in H$ and $w > 0$ satisfy $U_B^{\zeta}(u) \leq 1/w$. Then
$B + \zeta \cdot J - w \, u u^*$ is strictly positive and
$\Psi_J \bigl( B + \zeta \cdot J - w \, u u^* \bigr) \leq \Psi_J(B)$.

(c) $\Psi_J(B)^{-1} \cdot J \preceq B$.

(d) Let $\Lambda > 0$ with $J \preceq \Lambda \cdot 1$, let $n \geq 1$, let
$s \geq 1/n$ with $\Tr_e J = \Lambda (n s^2 + 1)$, put $\zeta = (1 + s)/n$ and
$d_0 = \zeta \Tr_e J / s$, let $u_1, \dots, u_n \in H$ and
$w_1, \dots, w_n \in \mathbb{R}$, and let
$B_n = d_0 \cdot 1 + n \zeta \cdot J - \sum_{i=1}^n w_i \, u_i u_i^*$. If $B_n$ is
strictly positive and $\Psi_J(B_n) \leq s / \zeta$, then
\[
  \sum_{i=1}^n w_i \, u_i u_i^* \; \preceq \; (1 + s)^2 \Lambda \cdot 1 .
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:operator-traces, lem:operator-rank-one-updates, lem:sqrt-conjugation, lem:barrier,
      lem:frame-bounds-from-potentials}
(a) The resolvent identity holds in any ring. The right-hand side is the trace of a
product of two positive operators, the first of them $J B^{-1} J$, which is again of
finite trace and positive; \Cref{lem:operator-traces}(b) gives the sign, and injectivity
of $J$ makes it strict.

(b) The inequality between real numbers is the one of \Cref{lem:barrier}; what has to
be redone is the closed form of the potential after a rank-one downdate, which rests on
\Cref{lem:operator-rank-one-updates}(a), and the positivity of the numerator, which rests
on the injectivity of $J$.

(c) The operator $B^{-1/2} J B^{-1/2}$ is positive and its trace along $e$ is again
$\Psi_J(B)$, so \Cref{lem:operator-traces}(a) bounds it by $\Psi_J(B) \cdot 1$, and
\Cref{lem:sqrt-conjugation} carries that back to $J \preceq \Psi_J(B) \cdot B$.

(d) As in \Cref{lem:frame-bounds-from-potentials}(b), with (c) in place of
\Cref{lem:inverse-trace-bounds}(b); the arithmetic is the same.
\end{proof}

\begin{lemma}[Averages along a countable second family]
\label{lem:countable-averages}
\lean{ContinuousLinearMap.nonneg_of_inner_eq_integral,
      ContinuousLinearMap.re_inner_eq_integral,
      ContinuousLinearMap.summable_re_inner_of_gram,
      ContinuousLinearMap.integral_re_inner_apply,
      Discretization.Infinite.integral_norm_sq_eq_traceAlong,
      Discretization.Infinite.integral_upperVerifier_lt,
      Discretization.Infinite.exists_admissible_point}
\leanok
\uses{def:potentials-and-verifiers, def:operator-upper-potential}
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let $H$
be a nonzero complex Hilbert space with a countable Hilbert basis
$e = (e_k)_{k \in \kappa}$. Let $a : \Omega \to \mathbb{C}^{\iota}$ be square-integrable in
every coordinate with $\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$ and $b : \Omega \to H$
square-integrable, let $J$ be the Gram operator of $b$, the bounded operator with
$\langle u, J u \rangle = \int_\Omega |\langle u, b(x) \rangle|^2 \, d\mu(x)$ for every
$u \in H$, and assume that $J$ is injective.

(a) $J$ is positive and of finite trace along $e$, and
$\Tr_e J = \int_\Omega \|b(x)\|^2 \, d\mu(x)$.

(b) For every positive bounded operator $Q$ on $H$,
\[
  \int_\Omega \re \bigl\langle b(x), Q \, b(x) \bigr\rangle \, d\mu(x) = \Tr_e (J Q).
\]

(c) Let $B$ be a strictly positive operator on $H$ and $\zeta > 0$. Then
\[
  \int_\Omega U_B^{\zeta} \bigl( b(x) \bigr) d\mu(x) < \frac{1}{\zeta} + \Psi_J(B).
\]

(d) Let moreover $A \in \mathbb{C}^{\iota \times \iota}$ be positive definite and
$0 < \delta < \Phi(A)^{-1}$ with $1/\zeta + \Psi_J(B) \leq 1/\delta - \Phi(A)$. Then
there is an $x \in \Omega$ with
$U_B^{\zeta}\bigl( b(x) \bigr) < L_A^{\delta}\bigl( a(x) \bigr)$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:hilbert-schmidt-adjoint, lem:operator-traces, lem:operator-barrier,
      lem:verifiers-on-average, lem:point-from-averages}
(a) The quadratic form $\langle u, J u \rangle$ is real and nonnegative, so $J$ is
positive. By Tonelli and \Cref{lem:hilbert-schmidt-adjoint}(a),
$\sum_k \langle e_k, J e_k \rangle = \sum_k \int_\Omega |\langle e_k, b \rangle|^2 \, d\mu
= \int_\Omega \|b\|^2 \, d\mu < \infty$, which is the finite trace and its value.

(b) Pointwise $\re \langle b, Q b \rangle = \|Q^{1/2} b\|^2$, which
\Cref{lem:hilbert-schmidt-adjoint}(a) turns into
$\sum_k |\langle Q^{1/2} e_k, b \rangle|^2$. Sum and integral may be exchanged because all
terms are nonnegative and their integrals sum to a finite value, and the Gram identity
turns the result into $\sum_k \re \langle Q^{1/2} e_k, J \, Q^{1/2} e_k \rangle$. That
equals $\Tr_e (J Q)$ because both are the squared Hilbert--Schmidt norm of
$Q^{1/2} J^{1/2}$, once read through the operator and once through its adjoint, by
\Cref{lem:hilbert-schmidt-adjoint}(b) and \Cref{lem:operator-traces}(b). Countability of
the index set of the basis enters here, in the interchange.

(c) By (b) the average is $\Tr_e (J X J X) / E + \Psi_J(B + \zeta \cdot J)$ with
$X = (B + \zeta \cdot J)^{-1}$ and $E$ the gain of \Cref{lem:operator-barrier}(a), and
$X \preceq B^{-1}$ bounds the numerator by $E/\zeta$.

(d) The average of the lower verifier exceeds $1/\delta - \Phi(A)$ by
\Cref{lem:verifiers-on-average}, which by assumption is at least $1/\zeta + \Psi_J(B)$,
which exceeds the average of the upper verifier by (c). These are two averages of real
functions, one coming from a matrix and one from an operator, so
\Cref{lem:point-from-averages}(a) applies unchanged.
\end{proof}

\section{The theorem}

\begin{theorem}[Sparsification for a Hilbert-space-valued second family]
\label{thm:operator-sparsification}
\lean{Discretization.Infinite.bss_generalized,
      Discretization.Infinite.bss_generalized_of_hilbertBasis,
      HilbertBasis.countable_of_separableSpace, HilbertBasis.nonempty_of_nontrivial,
      Discretization.Infinite.bss_generalized_of_gram_eq_one',
      Discretization.Infinite.upperState, Discretization.Infinite.exists_points_weights,
      Discretization.Infinite.bss_generalized_of_gram_eq_one,
      Discretization.Infinite.bss_generalized_of_unique,
      Discretization.Infinite.bss_generalized_of_small_dim,
      Discretization.Infinite.bss_generalized_of_unique_of_small_dim}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let $H$
be a nonzero separable complex Hilbert space. Let $a : \Omega \to \mathbb{C}^{\iota}$ be
square-integrable in every coordinate with $I = \int_\Omega a(x) a(x)^* \, d\mu(x)$
positive definite and $b : \Omega \to H$ square-integrable, let $J$ be the Gram operator of
$b$, the bounded operator with
$\langle u, J u \rangle = \int_\Omega |\langle u, b(x) \rangle|^2 \, d\mu(x)$ for every
$u \in H$, and assume that $J$ is injective. Let $\Lambda > 0$ satisfy
$J \preceq \Lambda \cdot 1$, and put $m = \#\iota$ and
$M = \int_\Omega \|b(x)\|^2 \, d\mu(x) / \Lambda$. Then for every $n \geq m$ there are points
$x_1, \dots, x_n \in \Omega$ and weights $w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot I
    \; \preceq \; \sum_{i=1}^n w_i \, a(x_i) a(x_i)^*
\]
in the matrix order and
\[
  \sum_{i=1}^n w_i \, b(x_i) b(x_i)^*
    \; \preceq \; \Bigl(1 + \sqrt{\tfrac{M-1}{n}}\Bigr)^2 \Lambda \cdot 1
\]
in the order of operators.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:two-barriers, prop:barrier-iteration, lem:barrier,
      lem:operator-barrier, lem:countable-averages, lem:frame-bounds-from-potentials,
      lem:point-from-averages, lem:operator-rank-one-updates, lem:gram-identities}
Choose a Hilbert basis $e = (e_k)_{k \in \kappa}$ of $H$. Its index set is countable: the
open balls of radius $1/2$ around the basis vectors are pairwise disjoint, because distinct
basis vectors are $\sqrt{2}$ apart, and a separable space holds only countably many
pairwise disjoint nonempty open sets. It is nonempty because $H \neq 0$. By
\Cref{lem:countable-averages}(a), $J$ is positive and of finite trace along $e$ with
$\Tr_e J = \int_\Omega \|b\|^2 \, d\mu$, so $M = \Tr_e J / \Lambda$; everything below works
along $e$, and is \texttt{Discretization.Infinite.bss\_generalized\_of\_hilbertBasis}.

The normalisation is removed as in the finite case: replace $a$ by
$I^{-1/2} a$, whose Gram matrix is $1$ by \Cref{lem:gram-identities}(b), and conjugate the
resulting lower bound by $I^{1/2}$; the upper bound does not involve the first family.
The case $I = 1$ is \texttt{Discretization.Infinite.bss\_generalized\_of\_gram\_eq\_one'},
and it splits into the same four cases as in the finite case. The
lower side is the finite argument verbatim, since the first family is finite in all of
them; only the upper side changes.

Case $m \geq 2$, $M \geq 1 + 1/n$
(\texttt{Discretization.Infinite.bss\_generalized\_of\_gram\_eq\_one}). The construction is
that of \Cref{prop:barrier-iteration} with the upper state the operator
$B_k = B_0 + k \zeta \cdot J - \sum_i w_i \, b(x_i) b(x_i)^*$: the step is
\Cref{lem:countable-averages}(d) followed by the two barrier lemmas, the lower one of
\Cref{lem:barrier} for matrices and \Cref{lem:operator-barrier}(b) for operators. The
initial data and the parameters are those of the proof of \Cref{thm:two-barriers}
verbatim. The lower read-off is \Cref{lem:frame-bounds-from-potentials}(a) and the upper
one \Cref{lem:operator-barrier}(d), and the constants come out the same.

Case $m = 1$, $M \geq 1 + 1/n$. As in the finite case, with
\Cref{lem:operator-barrier}(b) and \Cref{lem:countable-averages}(c) on the upper side and
\Cref{lem:operator-barrier}(d) for the upper bound. The lower verifier is the constant
$L(x) = n |a(x)|^2$, of average $n$, so the induction tracks the operator alone and the
largest admissible weight $1/w_i = L(x_i)$ makes the lower bound the identity
$\sum_i w_i |a(x_i)|^2 = 1$.

Case $m \geq 2$, $M \leq 1 + 1/n$. The induction of the finite case verbatim: it consumes
the second family only through the constant upper verifier, here
$U(x) = n \|b(x)\|^2 / \Tr_e J$, whose average is $n$ by \Cref{lem:countable-averages}(a).
The weights it produces satisfy $w_i \|b(x_i)\|^2 \leq \Tr_e J / n$, so
\Cref{lem:operator-rank-one-updates}(b) gives
$\sum_i w_i \, b(x_i) b(x_i)^* \preceq \Tr_e J \cdot 1 = M \Lambda \cdot 1$, and
$M \leq 1 + 1/n$ forces $M \leq (1 + \sqrt{(M-1)/n})^2$.

Case $m = 1$, $M \leq 1 + 1/n$. Both verifiers are the constants $L(x) = n |a(x)|^2$ and
$U(x) = n \|b(x)\|^2 / \Tr_e J$, of equal average $n$, so
\Cref{lem:point-from-averages}(b) produces a single point $y$ with $U(y) \leq L(y)$ and
$L(y) > 0$; using it $n$ times with the weight $w = 1/L(y)$ makes the lower bound the
identity, and $w \, U(y) \leq 1$ with \Cref{lem:operator-rank-one-updates}(b) gives the
upper bound.
\end{proof}

Along every Hilbert basis $e$ of $H$ the effective dimension is also $\Tr_e J / \Lambda$,
by \Cref{lem:countable-averages}(a); for a positive operator the trace does not depend on
the basis in any case (\Cref{lem:operator-traces}(c)).

\section{A countable family}

\begin{lemma}[A countable family as a map into $\ell_2$]
\label{lem:family-gram-operator}
\lean{Discretization.Family.ae_mem_l2, Discretization.Family.exists_measurableSet_mem_l2,
      Discretization.Family.toLp, Discretization.Family.memLp_toLp,
      Discretization.Family.gramOp, Discretization.Family.inner_gramOp_self,
      Discretization.Family.gramOp_nonneg, Discretization.Family.traceAlong_gramOp,
      Discretization.Family.gramOp_injective, Discretization.Family.gramOp_le,
      Discretization.Family.integral_comap_val, Discretization.Family.gram_comap_val}
\leanok
\uses{def:trace-along-basis}
Let $(\Omega, \mu)$ be a measure space, let $\kappa$ be a countable set, and let
$b = (b_p)_{p \in \kappa}$ be square-integrable functions on $\Omega$ with
$\sum_p \|b_p\|_{L_2}^2 < \infty$.

(a) There is a measurable set $S \subseteq \Omega$ of full measure with
$\sum_p |b_p(x)|^2 < \infty$ for every $x \in S$.

(b) If $\sum_p |b_p(x)|^2 < \infty$ for every $x \in \Omega$, then
$x \mapsto b(x) = (b_p(x))_p$ is a square-integrable map into $\ell_2(\kappa)$, and its
Gram operator $J = \int_\Omega b(x) b(x)^* \, d\mu(x)$ satisfies
\[
  \langle c, J c \rangle = \int_\Omega \Bigl| \sum_p \overline{c_p} \, b_p(x) \Bigr|^2 d\mu(x)
  \qquad \text{for every } c \in \ell_2(\kappa).
\]
It is positive, its trace along the standard basis $e$ of $\ell_2(\kappa)$ is
$\Tr_e J = \sum_p \|b_p\|_{L_2}^2$, it is injective if the right-hand side vanishes only
for $c = 0$, and $J \preceq \Lambda \cdot 1$ if the right-hand side is at most
$\Lambda \|c\|^2$ for every $c$.
\end{lemma}

\begin{proof}
\leanok
(a) By Tonelli, $\int_\Omega \sum_p |b_p|^2 \, d\mu = \sum_p \|b_p\|^2 < \infty$, so the
sum is finite almost everywhere, and the complement of a measurable null set containing
the exceptional points is a set $S$ as required.

(b) The finite partial sums $\sum_{p \in F} b_p(x) e_p$ converge to $b(x)$ in
$\ell_2(\kappa)$, so $b$ is strongly measurable, and $\int_\Omega \|b\|^2 \, d\mu =
\sum_p \|b_p\|^2$ by Tonelli. The integrand $b(x) b(x)^*$ has norm $\|b(x)\|^2$, so the
Bochner integral $J$ exists and commutes with pairing against $c$; since
$\langle c, b(x) b(x)^* c \rangle = |\langle c, b(x) \rangle|^2$ this gives the identity,
and with it positivity. On the diagonal, $\langle e_p, b(x) \rangle = b_p(x)$ gives
$\langle e_p, J e_p \rangle = \|b_p\|^2$. If $J c = 0$ the quadratic form vanishes at $c$,
which gives injectivity, and a bound on the quadratic form of the self-adjoint operator
$\Lambda \cdot 1 - J$ is positivity of that operator.
\end{proof}

Restricting the measure space to the set $S$ of (a) changes no integral and no Gram
matrix, because $S$ has full measure, and it makes (b) applicable. This is how the proof of
\Cref{thm:sparsification} chooses its points inside $S$: the construction runs in $S$.

\begin{remark}[What is left]
\label{rem:left}
\leanok
The number of points in \Cref{thm:sparsification} is governed by the effective dimension
$M = \Tr J / \Lambda$ and not by the size of the second family. What is not formalised is
the construction of the Gram matrix from a reproducing kernel: the singular value
decomposition of the embedding $H \hookrightarrow L_2(\mu)$, which supplies the family
$b$ and makes $J$ diagonal, with $\Tr J$ the trace of the kernel and $\Lambda$ the
squared norm of the embedding.
\end{remark}

\chapter{Discretizing the uniform norm}
\label{ch:design}

The sparsification theorem starts from a measure; this chapter produces one. Let $D$ be
an arbitrary set, $\iota$ a finite nonempty index set with $m = \#\iota$ elements, and
$a = (a_k)_{k \in \iota}$ a family of bounded functions $D \to \mathbb{C}$ whose
coordinate functions are linearly independent. For every $\varepsilon > 0$ we look for
finitely many points $x_k \in D$ with weights $w_k \geq 0$ summing to one for which
\[
  |f(y)|^2 \; \leq \; (m + \varepsilon) \sum_k w_k |f(x_k)|^2
  \qquad \text{for all } y \in D \text{ and all } f \in \operatorname{span}\{a_k\},
\]
that is, the uniform norm is dominated by the $L_2$ norm of that measure with the
constant $\sqrt{m + \varepsilon}$. The answer of Kiefer and Wolfowitz is that a measure
whose Gram matrix has an almost maximal determinant does this.

No topology and no measurable structure on $D$ are used: a finitely supported measure is
a finite list of points with weights, and it is kept in that form until
\Cref{cor:design-as-measure}.

A \emph{design} consists of finitely many points $x_1, \dots, x_N \in D$ and weights
$w_1, \dots, w_N \geq 0$ with $\sum_k w_k = 1$, and its \emph{Gram matrix} is
$G = \sum_k w_k \, a(x_k) a(x_k)^*$, the Gram matrix of $a$ in $L_2$ of the measure
$\sum_k w_k \delta_{x_k}$; in Lean it is
\texttt{Discretization.KieferWolfowitz.designGram}. The statements below spell both out.

\begin{lemma}[Quadratic form of the Gram matrix of a design]
\label{lem:design-quadratic-form}
\lean{Discretization.KieferWolfowitz.designGram,
      Discretization.KieferWolfowitz.re_quadForm_designGram}
\leanok
Let $D$ be a set, let $\iota$ be a finite set and let $a : D \to \mathbb{C}^{\iota}$. For
points $x_1, \dots, x_N \in D$, weights $w_1, \dots, w_N \geq 0$ with $\sum_k w_k = 1$
and $G = \sum_k w_k \, a(x_k) a(x_k)^*$, and for every $c \in \mathbb{C}^{\iota}$, one has
$c^* G c = \sum_k w_k |\langle c, a(x_k) \rangle|^2$.
\end{lemma}

\begin{proof}
\leanok
The quadratic form of a rank-one matrix $u u^*$ at $c$ is $|\langle c, u \rangle|^2$,
and both sides are additive in the summands.
\end{proof}

\begin{lemma}[Gain from mixing in one point]
\label{lem:mixing-gain}
\lean{Discretization.KieferWolfowitz.exists_mix_ge}
\leanok
Let $m \geq 1$ and $t > m$ be real. Then $\alpha = (t-m) / (2m(t-1))$ lies in $[0,1)$ and
\[
  1 + \frac{(t-m)^2}{4m(t-1)}
    \; \leq \; (1-\alpha)^{m-1} \bigl( 1 + \alpha (t-1) \bigr).
\]
\end{lemma}

\begin{proof}
\leanok
By Bernoulli's inequality $(1-\alpha)^{m-1} \geq 1 - (m-1)\alpha$, and the difference of
the two sides of the claim is then the explicit nonnegative quantity
$(t-m)^2 / (4m^2(t-1))$. The weight $\alpha$ is the one for which the linear gain
$\alpha(t-m)$ is twice the quadratic loss $(m-1)(t-1)\alpha^2$, so half of the gain
survives.
\end{proof}

\begin{proposition}[Almost maximal determinant]
\label{prop:almost-maximal-determinant}
\lean{Discretization.KieferWolfowitz.exists_design_quadForm_inv_le,
      Discretization.KieferWolfowitz.exists_design_posDef}
\leanok
Let $\iota$ be a finite nonempty index set with $m = \#\iota$ elements, let
$a = (a_k)_{k \in \iota}$ be bounded functions on a set $D$ whose coordinate functions
are linearly independent, and let $\varepsilon > 0$. Then there are points
$x_1, \dots, x_N \in D$ and weights $w_1, \dots, w_N \geq 0$ with $\sum_k w_k = 1$ such
that $G = \sum_k w_k \, a(x_k) a(x_k)^*$ is positive definite and
\[
  a(y)^* G^{-1} a(y) \; \leq \; m + \varepsilon \qquad \text{for all } y \in D .
\]
\end{proposition}

\begin{proof}
\leanok
\uses{lem:design-quadratic-form, lem:rank-one-updates, lem:mixing-gain}
First, some design has a positive definite Gram matrix
(\texttt{Discretization.KieferWolfowitz.exists\_design\_posDef}). Linear independence of
the coordinate functions says that $c = 0$ is the only coefficient vector with
$\langle c, a(y) \rangle = 0$ for all $y \in D$. It therefore suffices to find points
annihilating no nonzero coefficient vector, for then the uniform distribution on them has
$c^* G c > 0$ for every $c \neq 0$ by \Cref{lem:design-quadratic-form}. Descending
induction on dimension supplies them: while some nonzero vector survives the points
chosen so far, the hypothesis gives a point at which it does not vanish, and adding that
point lowers the dimension.

The determinants of the Gram matrices of designs form a set of real numbers which is
bounded above, since the entries of every Gram matrix are bounded by $C^2$ and the
Leibniz formula bounds a determinant by its entries, and which contains a positive
number by the first step. Let $S$ be its supremum and take a design whose
determinant exceeds $S / (1 + \eta)$, where
$\eta = \varepsilon^2 / (4m(m + \varepsilon - 1))$; its Gram matrix $G$ is positive
definite because its determinant is positive.

Appending a point $y$ with weight $\alpha \in [0,1)$ and scaling the old weights by
$1 - \alpha$ is again a design, with Gram matrix $(1-\alpha) G + \alpha \, a(y) a(y)^*$.
By \Cref{lem:rank-one-updates}(c) its determinant is $\det G$ multiplied by
$(1-\alpha)^{m-1}(1 + \alpha(t-1))$ with $t = a(y)^* G^{-1} a(y)$, and that multiple
cannot exceed $S / \det G < 1 + \eta$. If $t$ were larger than $m + \varepsilon$, the
weight of \Cref{lem:mixing-gain} would make it at least $1 + \eta$.
\end{proof}

\begin{theorem}[Discretization of the uniform norm]
\label{thm:uniform-discretization}
\lean{Discretization.KieferWolfowitz.exists_design_kieferWolfowitz}
\leanok
Let $\iota$ be a finite nonempty index set with $m = \#\iota$ elements, let
$a = (a_k)_{k \in \iota}$ be bounded functions on a set $D$ with linearly independent
coordinate functions, and let $\varepsilon > 0$. Then there are points
$x_1, \dots, x_n \in D$ and weights $w_1, \dots, w_n \geq 0$ with $\sum_k w_k = 1$ such
that
\[
  |f(y)|^2 \; \leq \; (m + \varepsilon) \sum_{k=1}^n w_k \, |f(x_k)|^2
\]
for every $y \in D$ and every $f \in \operatorname{span}\{a_k\}$.
\end{theorem}

\begin{proof}
\leanok
\uses{prop:almost-maximal-determinant, lem:design-quadratic-form, lem:rank-one-updates}
Take the design of \Cref{prop:almost-maximal-determinant} for $\varepsilon/2$. For $s = m + \varepsilon$
the bound $s^{-1} a(y)^* G^{-1} a(y) < 1$ is strict, so subtracting $s^{-1} a(y) a(y)^*$
from $G$ leaves a positive definite matrix by \Cref{lem:rank-one-updates}(b), that is
$a(y) a(y)^* \preceq s \cdot G$. Evaluating this at the coefficient vector of $f$ and
using \Cref{lem:design-quadratic-form} gives the claim.
\end{proof}

\begin{theorem}[The sharp constant on a compact domain]
\label{thm:uniform-discretization-compact}
\lean{Discretization.KieferWolfowitz.exists_design_kieferWolfowitz_of_compact}
\leanok
Let $\iota$ be a finite nonempty index set with $m = \#\iota$ elements, let $D$ be a
compact space and let $a = (a_k)_{k \in \iota}$ be continuous functions on it with
linearly independent coordinate functions. Then there are points
$x_1, \dots, x_n \in D$ and weights $w_1, \dots, w_n \geq 0$ with $\sum_k w_k = 1$ such
that
\[
  |f(y)|^2 \; \leq \; m \sum_{k=1}^n w_k \, |f(x_k)|^2
\]
for every $y \in D$ and every $f \in \operatorname{span}\{a_k\}$.
\end{theorem}

\begin{proof}
\leanok
\uses{prop:almost-maximal-determinant, lem:mixing-gain, lem:design-quadratic-form}
The matrices $a(y) a(y)^*$ form a compact set, whose convex hull is compact as well and
consists exactly of the Gram matrices of designs. The determinant is continuous, so it
attains its maximum at a design. No mixture can then increase the determinant at all,
and \Cref{lem:mixing-gain} applied with the factor $1$ gives $a(y)^* G^{-1} a(y) \leq m$ without
slack. The passage to the norm inequality is that of
\Cref{thm:uniform-discretization} for every constant $s > m$, followed by the limit
$s \to m$, which is a limit of real numbers.
\end{proof}

Both theorems produce a list of points with weights. Read as a measure, that list is
the object the sparsification theorem asks for.

\begin{corollary}[The design as a measure]
\label{cor:design-as-measure}
\lean{Discretization.KieferWolfowitz.exists_probabilityMeasure_kieferWolfowitz}
\leanok
Let $\iota$ be a finite nonempty index set with $m = \#\iota$ elements, let $D$ carry a
measurable structure, let $a = (a_k)_{k \in \iota}$ be bounded measurable functions on it
with linearly independent coordinate functions, and let $\varepsilon > 0$. Then there is
a finitely supported probability measure $\varrho = \sum_k w_k \delta_{x_k}$ on $D$ whose
Gram matrix $\int_D a(y) a(y)^* \, d\varrho(y)$ is positive definite and equal to
$\sum_k w_k \, a(x_k) a(x_k)^*$, and
\[
  |f(y)|^2 \; \leq \; (m + \varepsilon) \int_D |f|^2 \, d\varrho
\]
for every $y \in D$ and every $f \in \operatorname{span}\{a_k\}$.
\end{corollary}

\begin{proof}
\leanok
\uses{thm:uniform-discretization}
This is \Cref{thm:uniform-discretization} read through the measure: integration against $\sum_k w_k \delta_{x_k}$ is the weighted sum over the points, for
every strongly measurable integrand.
\end{proof}

\begin{lemma}[Few points suffice]
\label{lem:few-design-points}
\lean{Discretization.KieferWolfowitz.exists_design_card_le}
\leanok
Let $D$ be a set, let $\iota$ be a finite set with $m = \#\iota$ elements and let
$a : D \to \mathbb{C}^{\iota}$. For all points $x_1, \dots, x_N \in D$ and weights
$w_1, \dots, w_N \geq 0$ with $\sum_k w_k = 1$ there are at most $2m^2 + 1$ points
$y_j \in D$ and weights $v_j \geq 0$ with $\sum_j v_j = 1$ and
$\sum_j v_j \, a(y_j) a(y_j)^* = \sum_k w_k \, a(x_k) a(x_k)^*$.
\end{lemma}

\begin{proof}
\leanok
The Gram matrix lies in the convex hull of the matrices $a(y) a(y)^*$, so by
Carath\'eodory's theorem it is a convex combination of affinely independent ones, and an
affinely independent family in the real vector space $\mathbb{C}^{\iota \times \iota}$,
whose real dimension is $2m^2$, has at most $2m^2 + 1$ members.
\end{proof}

\begin{remark}[What the count is good for]
\label{rem:point-count}
\leanok
\uses{lem:few-design-points, thm:uniform-discretization, thm:uniform-discretization-compact,
      lem:design-quadratic-form}
Both \Cref{thm:uniform-discretization} and \Cref{thm:uniform-discretization-compact}
depend on the design only through its Gram matrix, by \Cref{lem:design-quadratic-form}, so
\Cref{lem:few-design-points} gives them with at most $2m^2 + 1$ points. The sharper count
$r + 1$, with $r$ the dimension of the span of the products $a_k \overline{a_l}$, is not
formalised, and nothing needs it.
\end{remark}

\begin{remark}[The identity decomposed over the design]
\label{rem:identity-decomposition}
\lean{Discretization.KieferWolfowitz.sum_weight_quadForm_inv_eq_card, Discretization.KieferWolfowitz.sum_weight_smul_quadForm_inv_eq_self}
\leanok
\uses{prop:almost-maximal-determinant, thm:uniform-discretization-compact}
For a design with points $x_1, \dots, x_n$, weights $w_1, \dots, w_n$ and invertible Gram
matrix $G$, and with $m = \#\iota$,
\[
  \sum_{k=1}^n w_k \, a(x_k)^* G^{-1} a(x_k) = m
  \qquad \text{and} \qquad
  \sum_{k=1}^n w_k \bigl( a(x_k)^* G^{-1} z \bigr) a(x_k) = z
\]
for every $z \in \mathbb{C}^{\iota}$, the first being $\Tr(G^{-1} G)$ and the second
$G G^{-1} z = z$ written out. With $c_k = w_k \, a(x_k)^* G^{-1} a(x_k)$ this is John's
decomposition of the identity over the design points, in the inner product
$\langle u, z \rangle = u^* G^{-1} z$ of the ellipsoid of $G$, and it is the condition
dual to the bound of \Cref{prop:almost-maximal-determinant}. Under
\Cref{thm:uniform-discretization-compact} the average forces
$a(x_k)^* G^{-1} a(x_k) = m$ at every design point, which makes the points contact
points. For an arbitrary convex body the theorem is proved in the companion project
\href{https://mario-ullrich.github.io/Lean-SNumbers/}{Lean-SNumbers}
(\href{https://mario-ullrich.github.io/Lean-SNumbers/blueprint.pdf}{PDF}).
\end{remark}
