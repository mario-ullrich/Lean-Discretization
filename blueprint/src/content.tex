\chapter{The problem and the result}
\label{ch:problem}

Let $(\Omega, \mu)$ be a measure space and $\iota$ a finite nonempty index set with
$m = \#\iota$ elements. Two objects are given: a finite family $a = (a_k)_{k \in \iota}$ of
square-integrable functions on $\Omega$, and a square-integrable map $b : \Omega \to H$
into a separable complex Hilbert space $H$. We ask whether the integral of $|f|^2$ can be
replaced by finitely many point evaluations, so that the first family stays a frame from
below and $b$ stays a frame from above. Two cases of $b$ are the ones to keep in mind.
\begin{itemize}
  \item A countable family $(b_k)_{k \in \kappa}$ of square-integrable functions, with
    $b(x) = (b_k(x))_k \in \ell_2(\kappa)$. The functions $\langle c, b(\cdot) \rangle$,
    for $c \in \ell_2(\kappa)$, form the span of the family.
  \item A reproducing kernel Hilbert space $H$ of functions on $\Omega$ with kernel $K$,
    and $b(x) = K(x, \cdot)$, the kernel section at $x$. Then
    $\langle b(x), f \rangle = f(x)$ for every $f \in H$.
\end{itemize}

For a family $a : \Omega \to \mathbb{C}^{\iota}$ that is square-integrable in every
coordinate, the \emph{Gram matrix} is
\[
  G(a) = \int_\Omega a(x) a(x)^* \, d\mu(x) \in \mathbb{C}^{\iota \times \iota},
\]
the matrix with entries $\int_\Omega a_k \overline{a_l} \, d\mu$; in Lean it is
\texttt{Discretization.gram}. The same formula defines the \emph{Gram operator}
\[
  J = \int_\Omega b(x) b(x)^* \, d\mu(x)
\]
of $b$, a Bochner integral of the rank-one operators
$b(x) b(x)^* = \langle b(x), \cdot \rangle \, b(x)$ on $H$. It is positive, with quadratic
form $\langle u, J u \rangle = \int_\Omega |\langle u, b(x) \rangle|^2 \, d\mu(x)$, and its
trace is $\Tr J = \int_\Omega \|b\|^2 \, d\mu$. For $\Lambda > 0$ with
$J \preceq \Lambda \cdot 1$, the number $M = \Tr J / \Lambda$ is the \emph{effective
dimension} of $J$ relative to $\Lambda$. For the family it is
$M = \sum_k \|b_k\|_{L_2}^2 / \Lambda$; for the kernel it is
$M = \int_\Omega K(x, x) \, d\mu(x) / \Lambda$, and $\Lambda$ bounds the squared norm of
the embedding $H \hookrightarrow L_2(\mu)$.

Then for every $n \geq m$ there are points $x_1, \dots, x_n \in \Omega$ and weights
$w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \int_\Omega |f|^2 \, d\mu
    \; \leq \; \sum_{i=1}^n w_i \, |f(x_i)|^2
  \qquad \text{for all } f \in \operatorname{span}\{a_k\},
\]
and
\[
  \sum_{i=1}^n w_i \, |\langle u, b(x_i) \rangle|^2
    \; \leq \; \Bigl(1 + \sqrt{\tfrac{M-1}{n}}\Bigr)^2 \Lambda \, \|u\|^2
  \qquad \text{for all } u \in H .
\]
This is \Cref{thm:discretization}, read through quadratic forms in
\Cref{cor:l2-discretization}. The two families enter asymmetrically: the first through its
size $m$, the second only through $M$, which is at most $\dim H$ and much smaller as soon
as the eigenvalues of $J$ decay. For the kernel the second bound reads
$\sum_i w_i |f(x_i)|^2 \leq (1 + \sqrt{(M-1)/n})^2 \Lambda \|f\|_H^2$
(\Cref{cor:rkhs-discretization}). Taking for $b$ the first family itself, with Gram matrix
the identity, gives $M = m$ and the sparsification theorem of Batson, Spielman and
Srivastava~\cite{BSS}, which is \Cref{cor:bss}.

The last chapter answers which measure to discretize in the first place. There $D$ is an
arbitrary set carrying $m$ bounded, linearly independent functions, and Kiefer and
Wolfowitz give, for every $\varepsilon > 0$, finitely many points and weights with
$|f(y)|^2 \leq (m + \varepsilon) \sum_k w_k |f(x_k)|^2$ for every $y \in D$ and every $f$
in their span: the uniform norm is dominated by the $L_2$ norm of that measure. Feeding
such a measure to \Cref{thm:discretization} is how one reaches sampling projections.

A statement about these weighted sums is a statement about the matrix
$\sum_i w_i \, a(x_i) a(x_i)^*$ and the operator $\sum_i w_i \, b(x_i) b(x_i)^*$, compared
in the Loewner order: $A \preceq B$ means that $B - A$ is positive semidefinite. That is the
form the argument takes. Traces appearing below are real, and the formalisation takes
real parts wherever a real number is needed; it writes a function of a span as
$f = \langle c, a(\cdot) \rangle$, which conjugates the coefficient vector, and since $c$
runs over all of $\mathbb{C}^{\iota}$ that is the same set of functions.

Each statement names the Lean declaration that formalises it. Every statement and every
proof is formalised: the project compiles without \texttt{sorry}, so each result is
verified by the Lean kernel.

The following theorem is Theorem 3 of Chkifa, Dolbeault, Krieg and Ullrich~\cite{CDKU}, for
a second family with values in a separable Hilbert space. Its first bound is their factor
$(1 - \sqrt{(m-1)/n})^2 \lambda_{\min}(I)$ in Loewner form.

\begin{theorem}[Discretization]
\label{thm:discretization}
\lean{Discretization.Infinite.bss_generalized, Discretization.bss_of_gram_eq_one,
      Discretization.Infinite.bss_generalized_of_gram_eq_one',
      Discretization.bss_generalized_of_gram_eq_one',
      Discretization.UpperBarrier.bss_generalized_of_gram_eq_one',
      Discretization.bss_generalized_of_gram_eq_one,
      Discretization.UpperBarrier.bss_generalized_of_gram_eq_one,
      Discretization.bss_generalized_of_unique,
      Discretization.UpperBarrier.bss_generalized_of_unique,
      Discretization.bss_generalized_of_small_dim,
      Discretization.UpperBarrier.bss_generalized_of_small_dim,
      Discretization.bss_generalized_of_unique_of_small_dim,
      Discretization.UpperBarrier.bss_generalized_of_unique_of_small_dim}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let $H$
be a nonzero separable complex Hilbert space. Let $a : \Omega \to \mathbb{C}^{\iota}$ be
square-integrable in every coordinate with $I = \int_\Omega a(x) a(x)^* \, d\mu(x)$
positive definite, let $b : \Omega \to H$ be square-integrable, and let
$J = \int_\Omega b(x) b(x)^* \, d\mu(x)$ be its Gram operator; assume that $J$ is
injective. Let $\Lambda > 0$ satisfy $J \preceq \Lambda \cdot 1$, and put $m = \#\iota$ and
$M = \int_\Omega \|b(x)\|^2 \, d\mu(x) / \Lambda$. Then for every $n \geq m$ there are points
$x_1, \dots, x_n \in \Omega$ and weights $w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot I
    \; \preceq \; \sum_{i=1}^n w_i \, a(x_i) a(x_i)^*
\]
in the matrix order and
\[
  \sum_{i=1}^n w_i \, b(x_i) b(x_i)^*
    \; \preceq \; \Bigl(1 + \sqrt{\tfrac{M-1}{n}}\Bigr)^2 \Lambda \cdot 1
\]
in the order of operators.
\end{theorem}

The proof is given in \Cref{ch:construction}, after the construction it rests on. No side
condition on $m$ or on $M$ is needed; the potential argument there needs two, and three
further cases of the proof remove them. In the formalisation the Gram operator enters as
the hypothesis $\int_\Omega b(x) b(x)^* \, d\mu(x) = J$, the counterpart of $G(b) = J$ for a
finite family.
Both hypotheses on $J$ can be read on $b$ (\Cref{lem:gram-operator}):
$J \preceq \Lambda \cdot 1$ is the bound
$\int_\Omega |\langle u, b \rangle|^2 \, d\mu \leq \Lambda \|u\|^2$ for every $u \in H$, and
$J$ is injective exactly when $\langle u, b(\cdot) \rangle = 0$ almost everywhere only for
$u = 0$. Equivalently, $J = T^* T$ for the operator $T : H \to L_2(\mu)$,
$u \mapsto \langle b(\cdot), u \rangle$, and $\Lambda$ bounds $\|T\|^2$.

For a countable family the theorem takes the form in which~\cite{CDKU} state their
Theorem 3, with the Gram matrix of the family in place of the Gram operator.

\begin{corollary}[Discretization for a countable second family]
\label{cor:countable-discretization}
\lean{Discretization.bss_generalized, Discretization.Countable.bss_generalized}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let
$\kappa$ be a nonempty set that is finite or countably infinite. Let
$a : \Omega \to \mathbb{C}^{\iota}$ be square-integrable in every coordinate with
$I = \int_\Omega a(x) a(x)^* \, d\mu(x)$ positive definite, and let
$b = (b_k)_{k \in \kappa}$ be square-integrable functions on $\Omega$ with
$\sum_k \|b_k\|_{L_2}^2 < \infty$. Let $J$ be their Gram matrix, with entries
$J_{kl} = \int_\Omega b_k \overline{b_l} \, d\mu$, as an operator on $\ell_2(\kappa)$;
assume that $J$ is injective, let $\Lambda > 0$ satisfy $J \preceq \Lambda \cdot 1$, and
put $m = \#\iota$ and $M = \Tr J / \Lambda = \sum_k \|b_k\|_{L_2}^2 / \Lambda$. Then for
every $n \geq m$ there are points $x_1, \dots, x_n \in \Omega$ with
$\sum_k |b_k(x_i)|^2 < \infty$ and weights $w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot I
    \; \preceq \; \sum_{i=1}^n w_i \, a(x_i) a(x_i)^*
\]
and
\[
  \sum_{i=1}^n w_i \, \Bigl| \sum_{k \in \kappa} c_k \, b_k(x_i) \Bigr|^2
    \; \leq \; \Bigl(1 + \sqrt{\tfrac{M-1}{n}}\Bigr)^2 \Lambda \sum_{k \in \kappa} |c_k|^2
  \qquad \text{for every } c \in \ell_2(\kappa).
\]
\end{corollary}

\begin{proof}
\leanok
\uses{thm:discretization, lem:countable-gram-operator}
For countably infinite $\kappa$, \Cref{lem:countable-gram-operator}(a) gives a measurable
set $S \subseteq \Omega$ of full measure on which $b(x) = (b_k(x))_k$ lies in
$\ell_2(\kappa)$. Restricted to $S$, which changes neither $I$ nor any of the integrals,
$b$ is a square-integrable map into the separable space $\ell_2(\kappa)$, with
$\int_S \|b\|^2 \, d\mu = \sum_k \|b_k\|^2$; its Gram operator is $J$, injective and bounded
by $\Lambda \cdot 1$ (\Cref{lem:countable-gram-operator}(b), (c)).
\Cref{thm:discretization} on $S$ gives the points,
which therefore lie in $S$, and the weights; its upper bound in the order of operators,
evaluated at $\bar c$, is the bound claimed.

For finite $\kappa$ the same argument works with $\mathbb{C}^{\kappa}$ in place of
$\ell_2(\kappa)$. The formalisation proves this case with the matrix form of the upper
side (the end of the proof of \Cref{thm:discretization}).
\end{proof}

In the formalisation the two hypotheses on $J$ are stated on the family, as in
\Cref{lem:countable-gram-operator}(c): $J \preceq \Lambda \cdot 1$ is the bound
$\int_\Omega |\sum_k \overline{c_k} \, b_k|^2 \, d\mu \leq \Lambda \|c\|^2$ for every
$c \in \ell_2(\kappa)$, and injectivity says that this integral vanishes only for $c = 0$.
For finite
$\kappa$ the corollary is \texttt{Discretization.bss\_generalized}, where injectivity is
positive definiteness and both finiteness conditions are automatic; for countable
$\kappa$ it is \texttt{Discretization.Countable.bss\_generalized}.

For $b = a$, \Cref{cor:countable-discretization} gives the sparsification theorem of Batson,
Spielman and Srivastava~\cite{BSS} in a slightly stronger form: $\sqrt{(m-1)/n}$ in place
of their $\sqrt{m/n}$, which comes from~\cite{CDKU}. Part (b) below is their statement for
finitely many vectors, over $\mathbb{C}$.

\begin{corollary}[Sparsification]
\label{cor:bss}
\lean{Discretization.bss, Discretization.bss_of_sum_eq_one, Discretization.gram_count,
      Discretization.sum_fiberwise_smul,
      Discretization.card_filter_sum_fiberwise_ne_zero_le}
\leanok
Let $\iota$ be a finite nonempty set and put $m = \#\iota$.

(a) Let $(\Omega, \mu)$ be a measure space and let $a : \Omega \to \mathbb{C}^{\iota}$ be
square-integrable in every coordinate with $\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$. Then
for every $n \geq m$ there are points $x_1, \dots, x_n \in \Omega$ and weights
$w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot 1
    \; \preceq \; \sum_{i=1}^n w_i \, a(x_i) a(x_i)^*
    \; \preceq \; \Bigl(1 + \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot 1 .
\]

(b) Let $Y$ be a finite set and let $v_y \in \mathbb{C}^{\iota}$, for $y \in Y$, be vectors
with $\sum_{y \in Y} v_y v_y^* = 1$. Then for every $n \geq m$ there are weights
$s_y \geq 0$, at most $n$ of them nonzero, with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot 1
    \; \preceq \; \sum_{y \in Y} s_y \, v_y v_y^*
    \; \preceq \; \Bigl(1 + \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \cdot 1 .
\]
\end{corollary}

\begin{proof}
\leanok
\uses{cor:countable-discretization}
(a) This is \Cref{cor:countable-discretization} for $b = a$ and $I = 1$: the Gram matrix of the
second family is then the identity, $\Lambda = 1$ bounds it, and its effective dimension is
$\Tr 1 / 1 = m$, so both frame constants are built from the same $r = \sqrt{(m-1)/n}$.

(b) The counting measure on $Y$ has Gram matrix $\sum_y v_y v_y^* = 1$, so (a) applies to
$y \mapsto v_y$. A vector chosen several times receives the sum of its weights,
$s_y = \sum_{i : x_i = y} w_i$, which leaves both sums unchanged, and a nonzero $s_y$ belongs
to one of the $n$ chosen points.
\end{proof}

With the constant $\sqrt{m/n}$ of~\cite{BSS} and $n = d\,m$, the ratio of the upper to
the lower constant, the condition number, is $\bigl((\sqrt{d}+1)/(\sqrt{d}-1)\bigr)^2$.

Read through quadratic forms, the theorem becomes a statement about norms, Corollary 4
of~\cite{CDKU}.

\begin{corollary}[Discretization of the $L_2$-norm]
\label{cor:l2-discretization}
\lean{Discretization.Infinite.exists_discretization, Discretization.exists_discretization,
      Discretization.Countable.exists_discretization,
      Discretization.mul_integral_norm_sq_le_sum,
      Discretization.Infinite.sum_mul_norm_sq_inner_le}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let $H$
be a nonzero separable complex Hilbert space. Let $a : \Omega \to \mathbb{C}^{\iota}$ be
square-integrable in every coordinate with $\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$, and let
$b : \Omega \to H$ be square-integrable with injective Gram operator
$J = \int_\Omega b(x) b(x)^* \, d\mu(x)$. Let $\Lambda > 0$ satisfy
$J \preceq \Lambda \cdot 1$, and put $m = \#\iota$ and
$M = \int_\Omega \|b(x)\|^2 \, d\mu(x) / \Lambda$. Then for every $n \geq m$ there are points
$x_1, \dots, x_n \in \Omega$ and weights $w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \int_\Omega |f|^2 \, d\mu
  \; \leq \; \sum_{i=1}^n w_i \, |f(x_i)|^2
\]
for every $f \in \operatorname{span}\{a_k\}$, and with
\[
  \sum_{i=1}^n w_i \, |\langle u, b(x_i) \rangle|^2
  \; \leq \; \Bigl(1 + \sqrt{\tfrac{M-1}{n}}\Bigr)^2 \Lambda \, \|u\|^2
\]
for every $u \in H$.
\end{corollary}

\begin{proof}
\leanok
\uses{thm:discretization, lem:gram-identities}
A Loewner inequality is an inequality between quadratic forms. For $f = \sum_k c_k a_k$ one
has $\sum_i w_i |f(x_i)|^2 = c^* G c$ with $G$ the accumulated matrix, and
$\int_\Omega |f|^2 d\mu = c^* G(a) c = \|c\|^2$ by \Cref{lem:gram-identities}(a) and
$G(a) = 1$. On the other side,
$\langle u, \sum_i w_i \, b(x_i) b(x_i)^* u \rangle = \sum_i w_i |\langle u, b(x_i) \rangle|^2$.
Both bounds are therefore those of \Cref{thm:discretization} with $I = 1$.
\end{proof}

For a countable family, with $b(x) = (b_k(x))_k$ and $u = c \in \ell_2(\kappa)$, the second
bound reads $\sum_i w_i |g(x_i)|^2 \leq (1 + \sqrt{(M-1)/n})^2 \Lambda \sum_k |c_k|^2$ for
$g = \sum_k c_k \, b_k$, with $M = \sum_k \|b_k\|_{L_2}^2 / \Lambda$
(\texttt{Discretization.exists\_discretization} for finite $\kappa$,
\texttt{Discretization.Countable.exists\_discretization} for countable $\kappa$). It is
stated for the coefficient vector, because its right-hand side is the coefficient norm and
the family is not assumed linearly independent, so a function of its span need not
determine the coefficients.

For the kernel sections of a reproducing kernel Hilbert space the second bound is a
discretization of the norm of the space itself.

\begin{corollary}[Discretization in a reproducing kernel Hilbert space]
\label{cor:rkhs-discretization}
\lean{Discretization.RKHS.exists_discretization}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let
$a : \Omega \to \mathbb{C}^{\iota}$ be square-integrable in every coordinate with
$\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$. Let $H$ be a nonzero separable complex Hilbert
space of measurable functions on $\Omega$ on which every point evaluation is continuous,
let $K$ be its reproducing kernel, and assume $\int_\Omega K(x, x) \, d\mu(x) < \infty$. Let
$\Lambda > 0$ satisfy $\int_\Omega |f|^2 \, d\mu \leq \Lambda \|f\|_H^2$ for every
$f \in H$, and assume that the only $f \in H$ vanishing almost everywhere is $f = 0$. Put
$m = \#\iota$ and $M = \int_\Omega K(x, x) \, d\mu(x) / \Lambda$. Then for every $n \geq m$
there are points $x_1, \dots, x_n \in \Omega$ and weights $w_1, \dots, w_n > 0$ with
\[
  \Bigl(1 - \sqrt{\tfrac{m-1}{n}}\Bigr)^2 \int_\Omega |g|^2 \, d\mu
  \; \leq \; \sum_{i=1}^n w_i \, |g(x_i)|^2
\]
for every $g \in \operatorname{span}\{a_k\}$, and with
\[
  \sum_{i=1}^n w_i \, |f(x_i)|^2
  \; \leq \; \Bigl(1 + \sqrt{\tfrac{M-1}{n}}\Bigr)^2 \Lambda \, \|f\|_H^2
\]
for every $f \in H$.
\end{corollary}

\begin{proof}
\leanok
\uses{cor:l2-discretization, lem:kernel-sections, lem:gram-operator}
Take for $b(x)$ the kernel section $k_x$ of \Cref{lem:kernel-sections}: $b$ is
square-integrable, $|\langle f, k_x \rangle| = |f(x)|$ and $\|k_x\|^2 = K(x, x)$. By
\Cref{lem:gram-operator}(b) and (c) the two assumptions on $H$ say that the Gram operator
of $b$ satisfies $J \preceq \Lambda \cdot 1$ and is injective, and
$\int_\Omega \|b\|^2 \, d\mu = \int_\Omega K(x, x) \, d\mu$. \Cref{cor:l2-discretization}
gives the claim.
\end{proof}

The constant $\Lambda$ is the squared norm of the embedding $H \hookrightarrow L_2(\mu)$, and
the assumption on vanishing functions says that this embedding is injective. The kernel
sections themselves are the second family, so no basis of $H$ and no singular value
decomposition of the embedding enter.

\chapter{Matrices in the Loewner order}
\label{ch:loewner}

The construction keeps a positive definite matrix above a multiple of the identity and
controls it through one real number. This chapter collects what is needed about the
Loewner order, the trace and rank-one updates of matrices; \Cref{ch:infinite} does the
same for operators. Nothing here mentions sampling; all of it is stated for a general
field $\mathbb{C}$ or $\mathbb{R}$ where the C$^*$-algebra structure is not needed, and
none of it is in Mathlib.

\section{Comparison with multiples of the identity}

\begin{lemma}[Comparison with multiples of the identity]
\label{lem:loewner-basic-bounds}
\lean{Matrix.IsHermitian.le_smul_one, Matrix.PosSemidef.le_trace_smul_one,
      Matrix.PosDef.inv_le_inv_of_le, Matrix.PosDef.smul_one_le_of_inv_le}
\leanok
Let $\iota$ be a finite set; all matrices below are in $\mathbb{C}^{\iota \times \iota}$.

(a) Let $A$ be Hermitian and $c$ a real number with $\lambda_i(A) \leq c$ for every
eigenvalue. Then $A \preceq c \cdot 1$.

(b) Every positive semidefinite $A$ satisfies $A \preceq (\Tr A) \cdot 1$.

(c) If $A$ and $B$ are positive definite with $A \preceq B$, then
$B^{-1} \preceq A^{-1}$.

(d) If $A$ is positive definite and $A^{-1} \preceq c \cdot 1$ for some $c > 0$, then
$c^{-1} \cdot 1 \preceq A$.
\end{lemma}

\begin{proof}
\leanok
(a) By the spectral theorem $c \cdot 1 - A$ is unitarily conjugate to the diagonal
matrix with entries $c - \lambda_i(A) \geq 0$, and conjugation by a unitary preserves
positive semidefiniteness.

(b) The trace is the sum of the eigenvalues of $A$, all of which are nonnegative, so
every single eigenvalue is at most $\Tr A$, and (a) applies with $c = \Tr A$.

(c), (d) Both are read off Mathlib's antitonicity of the inverse in a C$^*$-algebra with
a star ordering, applied to the units of $\mathbb{C}^{\iota \times \iota}$ with the
$\ell_2$-operator norm; the inverse of $c \cdot 1$ is $c^{-1} \cdot 1$.
\end{proof}

\begin{lemma}[Conjugation by the square root]
\label{lem:sqrt-conjugation}
\lean{CStarAlgebra.inv_smul_le_of_conj_inv_sqrt_le}
\leanok
Let $A$ be a C$^*$-algebra with a star ordering, let $B \in A$ be positive and
invertible, let $X \in A$ and let $t > 0$. If
$B^{-1/2} X B^{-1/2} \preceq t \cdot 1$, then $t^{-1} \cdot X \preceq B$.
\end{lemma}

\begin{proof}
\leanok
Conjugating the hypothesis by $B^{1/2}$ preserves the order. On the left the two
inverse square roots cancel, on the right $t \cdot 1$ becomes $t \cdot B$; dividing by
$t$ gives the claim.
\end{proof}

The next lemma is the reason the whole argument can be run on two real numbers: a bound
on the trace of an inverse is a bound on the matrix itself.

\begin{lemma}[Bounds from the trace of an inverse]
\label{lem:inverse-trace-bounds}
\lean{Matrix.PosDef.inv_re_trace_smul_one_le, Matrix.PosDef.sub_smul_one}
\leanok
Let $\iota$ be a finite set and let $A \in \mathbb{C}^{\iota \times \iota}$ be positive
definite.

(a) The matrix $A$ satisfies
\[
  \bigl( \Tr A^{-1} \bigr)^{-1} \cdot 1 \; \preceq \; A .
\]

(b) If $0 < \delta < (\Tr A^{-1})^{-1}$, then $A - \delta \cdot 1$ is positive definite.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:loewner-basic-bounds}
(a) \Cref{lem:loewner-basic-bounds}(b) applied to $A^{-1}$ gives
$A^{-1} \preceq (\Tr A^{-1}) \cdot 1$, and \Cref{lem:loewner-basic-bounds}(d) turns that
around.

(b) Write
$A - \delta \cdot 1 = \bigl( A - (\Tr A^{-1})^{-1} \cdot 1 \bigr)
  + \bigl( (\Tr A^{-1})^{-1} - \delta \bigr) \cdot 1$, a positive semidefinite matrix by
(a) plus a positive definite one.
\end{proof}

\section{Traces of products}

\begin{lemma}[Traces of products]
\label{lem:trace-inequalities}
\lean{Matrix.PosSemidef.trace_mul_nonneg, Matrix.PosDef.re_trace_mul_pos,
      Matrix.PosSemidef.norm_trace_mul_sq_le, Matrix.PosSemidef.re_trace_mul_sq_le}
\leanok
Let $\iota$ be a finite set; all matrices below are in $\mathbb{C}^{\iota \times \iota}$.

(a) For positive semidefinite $P$ and $Q$ we have $\Tr (P Q) \geq 0$. If $P$ is
positive definite, $Q$ is positive semidefinite and $Q \neq 0$, then
$\Tr (P Q) > 0$.

(b) Let $Y$ be positive semidefinite and $Z$ any matrix. Then
\[
  \bigl| \Tr (Z Y) \bigr|^2 \; \leq \; \Tr Y \cdot \Tr (Z Y Z^*) .
\]
\end{lemma}

\begin{proof}
\leanok
(a) Write $Q$ as a sum of rank-one matrices $v v^*$. Then
$\Tr (P v v^*) = v^* P v \geq 0$, with strict inequality for $v \neq 0$ when $P$ is
positive definite, and at least one of the $v$ is nonzero when $Q \neq 0$.

(b) A positive semidefinite $Y$ makes $\langle P, Q \rangle = \Tr (Q Y P^*)$ a
semi-inner product on matrices. Cauchy--Schwarz for the pair $1$, $Z$ is the
inequality. No commutation of $Y$ and $Z$ is used.
\end{proof}

\section{Rank-one updates}

\begin{lemma}[Rank-one updates: Sherman--Morrison and the determinant]
\label{lem:rank-one-updates}
\lean{Matrix.inv_add_smul_vecMulVec, Matrix.trace_inv_add_smul_vecMulVec,
      Matrix.PosDef.add_smul_vecMulVec, Matrix.PosDef.sub_smul_vecMulVec,
      Matrix.det_smul_add_smul_vecMulVec}
\leanok
Let $\iota$ be a finite set with $m = \#\iota$ elements and let
$v \in \mathbb{C}^{\iota}$.

(a) Let $A$ be Hermitian and invertible and $t$ a scalar with
$1 + t \, v^* A^{-1} v \neq 0$. Then $A + t \, v v^*$ is invertible,
\[
  \bigl( A + t \, v v^* \bigr)^{-1}
    = A^{-1} - \frac{t}{1 + t \, v^* A^{-1} v} \,
      \bigl( A^{-1} v \bigr) \bigl( A^{-1} v \bigr)^* ,
\]
and consequently
\[
  \Tr \bigl( A + t \, v v^* \bigr)^{-1}
    = \Tr A^{-1} - \frac{t}{1 + t \, v^* A^{-1} v} \, v^* A^{-2} v .
\]

(b) Let $A$ be positive definite and $w \geq 0$. Then $A + w \, v v^*$ is positive
definite. If moreover $w \, v^* A^{-1} v < 1$, then $A - w \, v v^*$ is positive
definite.

(c) Let $A$ be invertible, $\beta \neq 0$ a scalar and $\alpha$ a scalar. Then
\[
  \det(\beta A + \alpha \, v v^*)
    = \beta^{\,m-1} \det A \, \bigl( \beta + \alpha \, v^* A^{-1} v \bigr).
\]
\end{lemma}

\begin{proof}
\leanok
(a) Multiplying the right-hand side by $A + t \, v v^*$ collapses, by Hermitian symmetry
of $A^{-1}$, to $1$ plus a multiple of $v (A^{-1} v)^*$ whose scalar factor vanishes by
the choice of the coefficient. Taking traces and using
$\Tr \bigl( (A^{-1} v)(A^{-1} v)^* \bigr) = v^* A^{-2} v$ gives the trace identity.

(b) The first is a positive definite matrix plus a positive semidefinite one. For the
second, (a) exhibits the inverse of $A - w \, v v^*$ as a positive definite matrix plus a
positive semidefinite one, and a matrix whose inverse is positive definite is itself
positive definite.

(c) Factor $\beta A$ out and apply the matrix determinant lemma
$\det(A + v w^*) = \det A \, (1 + w^* A^{-1} v)$, which holds because the added matrix
has rank one. The dependence on $\alpha$ is affine; this identity is what replaces the
derivative of the determinant.
\end{proof}

\chapter{Averages of quadratic forms}
\label{ch:averages-of-forms}

The bridge between the analytic side of the problem and the matrix algebra is one
identity: the average of a quadratic form along a family is a trace against its Gram
matrix. Read from right to left it says that a trace is an average, which is how the
construction finds a point.

\begin{lemma}[Two identities for the Gram matrix]
\label{lem:gram-identities}
\lean{Discretization.gram, Discretization.integral_quadForm,
      Discretization.integral_re_quadForm, Discretization.gram_mulVec}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite set, let
$a : \Omega \to \mathbb{C}^{\iota}$ be square-integrable in every coordinate, and write
$G(a) = \int_\Omega a(x) a(x)^* \, d\mu(x)$.

(a) For every matrix $Q \in \mathbb{C}^{\iota \times \iota}$,
\[
  \int_\Omega a(x)^* Q \, a(x) \, d\mu(x) = \Tr \bigl( Q \, G(a) \bigr).
\]

(b) For every matrix $S \in \mathbb{C}^{\iota \times \iota}$,
\[
  G(S a) = S \, G(a) \, S^* .
\]
\end{lemma}

\begin{proof}
\leanok
(a) Expand both sides as a double sum over $\iota$ and exchange the finite sum with the
integral. The integrand is integrable by Cauchy--Schwarz for $L_2$-functions.

(b) Expand both sides entrywise and exchange the finite sums with the integral.
\end{proof}

\begin{lemma}[A good point from two averages]
\label{lem:point-from-averages}
\lean{Discretization.exists_lt_of_integral_lt, Discretization.exists_le_of_integral_le}
\leanok
Let $(\Omega, \mu)$ be a measure space and let $f, g : \Omega \to \mathbb{R}$ be
integrable.

(a) If $\int_\Omega g \, d\mu < \int_\Omega f \, d\mu$, then $g(x) < f(x)$ for at least
one $x \in \Omega$.

(b) If $f$ and $g$ are nonnegative with $\int_\Omega g \, d\mu \leq \int_\Omega f \, d\mu$
and $\int_\Omega f \, d\mu > 0$, then there is an $x \in \Omega$ with $g(x) \leq f(x)$
and $f(x) > 0$.
\end{lemma}

\begin{proof}
\leanok
(a) If $f \leq g$ everywhere then $\int f \leq \int g$.

(b) If $f \leq g$ everywhere then the two averages agree, so the nonnegative function
$g - f$ has average zero and vanishes almost everywhere. At every point where $f$ and
$g$ agree, the assumption forces $f$ to vanish, so $f$ vanishes almost everywhere,
contradicting $\int f > 0$.
\end{proof}

The paper states step (a) as: the set of admissible points has positive measure. One
point is all the construction uses. When the two averages are equal, which happens in
the doubly degenerate case in the proof of \Cref{thm:discretization}, a strict
inequality is no longer available and (b) takes over. Both the nonnegativity of $g$ and the conclusion $f(x) > 0$ in (b)
are needed: for $f = (1, 0)$ and $g = (2, -1)$ on two points with the counting measure
both averages are $1$, and the only point with $g \leq f$ has $f = 0$.

\chapter{The lower potential and its barrier}
\label{ch:barrier}

The first family is carried through the construction by a positive definite matrix $A$,
which must stay above a multiple of the identity. It is controlled by a single real
number, and whether a candidate point may be used is decided by a function of that point.
The second family is treated in the same way by an operator, in \Cref{ch:infinite}.

\section{The potential and the verifier}

\begin{definition}[The lower potential and verifier]
\label{def:lower-potential}
\lean{Discretization.lowerPotential, Discretization.lowerVerifier}
\leanok
Let $\iota$ be a finite set, let $A \in \mathbb{C}^{\iota \times \iota}$ be positive
definite, and let $\delta > 0$ with $A - \delta \cdot 1$ positive definite. The
\emph{lower potential} of $A$ is $\Phi(A) = \Tr A^{-1}$. With
$Z = (A - \delta \cdot 1)^{-1}$, the \emph{lower verifier} at $v \in \mathbb{C}^{\iota}$ is
\[
  L_A^{\delta}(v)
    = \frac{v^* Z^2 v}{\Phi(A - \delta \cdot 1) - \Phi(A)} - v^* Z v .
\]
\end{definition}

The lower potential is large when $A$ has a small eigenvalue. Before each step $A$ is
shrunk by $\delta \cdot 1$, which raises the potential by an amount that is computed
exactly, and that amount is the budget one new sampling point is allowed to consume. The
verifier decides pointwise, without recomputing any inverse, whether a candidate point
stays within that budget. The upper potential $\Psi_J(B) = \Tr (J B^{-1})$ and its verifier
play the same role for the second family (\Cref{def:upper-potential}).

\begin{lemma}[How the lower potential changes]
\label{lem:lower-potential-change}
\lean{Discretization.lowerPotential_sub_eq, Discretization.lowerPotential_add_smul_vecMulVec}
\leanok
\uses{def:lower-potential}
Let $\iota$ be a finite set.

(a) Let $A \in \mathbb{C}^{\iota \times \iota}$ be positive definite and let $\delta > 0$
with $A - \delta \cdot 1$ positive definite. Then
\[
  \Phi(A - \delta \cdot 1) - \Phi(A)
    = \delta \, \Tr \bigl( (A - \delta \cdot 1)^{-1} A^{-1} \bigr) > 0 .
\]

(b) Let $N \in \mathbb{C}^{\iota \times \iota}$ be positive definite,
$v \in \mathbb{C}^{\iota}$ and $w \geq 0$. Then
\[
  \Phi \bigl( N + w \, v v^* \bigr)
    = \Phi(N) - \frac{w}{1 + w \, v^* N^{-1} v} \, v^* N^{-2} v .
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:trace-inequalities, lem:rank-one-updates}
(a) The identity is the resolvent identity $X^{-1} - Y^{-1} = X^{-1} (Y - X) Y^{-1}$ for
the matrices $X = A - \delta \cdot 1$ and $Y = A$, whose difference is $\delta \cdot 1$. The
sign is \Cref{lem:trace-inequalities}(a) applied to the product of two positive definite
matrices.

(b) Insert the Sherman--Morrison formula of \Cref{lem:rank-one-updates}(a) and take
traces.
\end{proof}

\begin{lemma}[Barrier lemma for the lower potential]
\label{lem:lower-barrier}
\lean{Discretization.lowerPotential_update_le}
\leanok
\uses{def:lower-potential}
Let $\iota$ be a finite set, let $A \in \mathbb{C}^{\iota \times \iota}$ be positive
definite, let $0 < \delta < \Phi(A)^{-1}$ and $v \in \mathbb{C}^{\iota}$, and let $w > 0$
satisfy $1/w \leq L_A^{\delta}(v)$. Then $A - \delta \cdot 1 + w \, v v^*$ is positive
definite and
\[
  \Phi \bigl( A - \delta \cdot 1 + w \, v v^* \bigr) \leq \Phi(A).
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:lower-potential-change, lem:rank-one-updates, lem:inverse-trace-bounds}
The matrix $A - \delta \cdot 1$ is positive definite by \Cref{lem:inverse-trace-bounds}(b),
and adding $w \, v v^*$ keeps it so by \Cref{lem:rank-one-updates}(b). The closed form of
\Cref{lem:lower-potential-change}(b) for $N = A - \delta \cdot 1$, together with the positive
gain of \Cref{lem:lower-potential-change}(a), reduces the inequality to one between real
numbers, which is the hypothesis $1/w \leq L_A^{\delta}(v)$ multiplied out.
\end{proof}

\section{The verifier passes on average}

A point may be used if the upper verifier stays below the lower one. Such a point exists
because the lower verifier is large on average.

\begin{lemma}[The lower verifier passes on average]
\label{lem:lower-verifier-on-average}
\lean{Discretization.integral_lowerVerifier_gt,
      Discretization.UpperBarrier.exists_admissible_point}
\leanok
\uses{def:lower-potential}
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let
$a : \Omega \to \mathbb{C}^{\iota}$ be square-integrable in every coordinate with
$\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$. Let $A \in \mathbb{C}^{\iota \times \iota}$ be
positive definite and $0 < \delta < \Phi(A)^{-1}$.

(a) The lower verifier satisfies
\[
  \int_\Omega L_A^{\delta}\bigl( a(x) \bigr) d\mu(x) > \frac{1}{\delta} - \Phi(A).
\]

(b) Let $U : \Omega \to \mathbb{R}$ be integrable with
$\int_\Omega U \, d\mu \leq 1/\delta - \Phi(A)$. Then there is an $x \in \Omega$ with
$U(x) < L_A^{\delta}\bigl( a(x) \bigr)$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:gram-identities, lem:trace-inequalities, lem:lower-potential-change,
      lem:point-from-averages}
(a) Write $Y = A^{-1}$ and $Z = (A - \delta \cdot 1)^{-1}$. \Cref{lem:gram-identities}(a)
turns the average into $\Tr (Z^2) / (\delta \Tr (Y Z)) - \Tr Z$, the denominator being the
gain of \Cref{lem:lower-potential-change}(a). Writing $t = \Tr (Y Z)$ and $u = \Tr (Y Z^2)$, the
claim reduces to $\delta t^2 < u$, which is what \Cref{lem:trace-inequalities}(b) gives
together with $\delta \Phi(A) < 1$.

(b) The lower verifier is a quadratic form in $a(x)$ and hence integrable. By (a) its
average exceeds $1/\delta - \Phi(A) \geq \int_\Omega U \, d\mu$, and
\Cref{lem:point-from-averages}(a) gives the point.
\end{proof}

In the construction $U$ is the upper verifier along the second family, whose average is
below $1/\zeta + \Psi_J(B)$ by \Cref{lem:upper-verifier-on-average}; the gap condition
$1/\zeta + \Psi_J(B) \leq 1/\delta - \Phi(A)$ then makes (b) applicable. Any weight $w$
between the two reciprocals is admissible for \Cref{lem:lower-barrier} and for
\Cref{lem:upper-barrier}, and the construction takes the largest one,
$w = 1/L_A^{\delta}$. Part (b) compares two averages of real functions and nothing else; in
Lean it is proved once for every upper state,
\texttt{Discretization.UpperBarrier.exists\_admissible\_point}.

\section{From the potential to the frame bound}

\begin{lemma}[From the potential to the lower frame bound]
\label{lem:lower-frame-bound}
\lean{Discretization.lower_bound_of_state, Discretization.lower_frame_bound}
\leanok
\uses{def:lower-potential}
Let $\iota$ be a finite nonempty set, let $n \geq 1$, and let
$v_1, \dots, v_n \in \mathbb{C}^{\iota}$ and $w_1, \dots, w_n \in \mathbb{R}$. Let $r > 0$
with $r \neq 1$, put $\delta = (1 - r)/n$ and $c_0 = \delta (n r^2 + 1)/r$, and let
$A_n = c_0 \cdot 1 - n \delta \cdot 1 + \sum_{i=1}^n w_i \, v_i v_i^*$. If $A_n$ is
positive definite and $\Phi(A_n) \leq r / \delta$, then
\[
  (1 - r)^2 \cdot 1 \; \preceq \; \sum_{i=1}^n w_i \, v_i v_i^* .
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:inverse-trace-bounds}
By \Cref{lem:inverse-trace-bounds}(a), $\Phi(A_n) \leq r/\delta$ gives
$(\delta / r) \cdot 1 \preceq A_n$; inserting the definition of $A_n$ gives
$(\delta/r + n\delta - c_0) \cdot 1 \preceq \sum_i w_i \, v_i v_i^*$, and
$\delta/r + n\delta - c_0 = n \delta (1 - r) = (1 - r)^2$. The arithmetic is in
\texttt{Discretization/Parameters.lean}.
\end{proof}

For $r = \sqrt{(m-1)/n}$ one has $n r^2 + 1 = m$; the lemma is stated in $r$ because its
proof never unfolds the square root.

\chapter{Operators on a Hilbert space}
\label{ch:infinite}

The upper half of the construction is carried out for a square-integrable map
$b : \Omega \to H$ into a separable Hilbert space, with Gram operator
$J = \int_\Omega b(x) b(x)^* \, d\mu(x)$. Nothing in the argument ties the number of points
to the size of the second family: it is governed by the effective dimension alone. A
finite second family is the case $H = \mathbb{C}^{\kappa}$, where every operator below is a
matrix and every trace a matrix trace. Mathlib has no trace-class theory, so the trace of a
positive operator is built here from a Hilbert basis.

The Gram operator is positive, since its quadratic form is nonnegative, and its trace is
$\int_\Omega \|b\|^2 \, d\mu$, by Parseval and monotone convergence
(\Cref{lem:gram-operator}(d)). The only genuine assumption on $J$ is therefore that it is
injective, which is what positive definiteness of the Gram matrix says for a finite family,
and \Cref{thm:discretization} assumes nothing else.

\section{The trace of a positive operator}

\begin{lemma}[Parseval and Hilbert--Schmidt sums under adjoints]
\label{lem:hilbert-schmidt-adjoint}
\lean{HilbertBasis.hasSum_norm_sq_inner, ContinuousLinearMap.tsum_norm_sq_adjoint,
      ContinuousLinearMap.summable_norm_sq_adjoint_iff,
      ContinuousLinearMap.tsum_ofReal_norm_sq_adjoint_of_basis,
      ContinuousLinearMap.tsum_ofReal_norm_sq_of_isSelfAdjoint}
\leanok
Let $H$ be a complex Hilbert space with a Hilbert basis $e = (e_k)_{k \in \kappa}$.

(a) For every $x \in H$ the family $\bigl( |\langle e_k, x \rangle|^2 \bigr)_k$ has sum
$\|x\|^2$.

(b) For every bounded operator $T$ on $H$,
\[
  \sum_k \| T e_k \|^2 = \sum_k \| T^* e_k \|^2 ,
\]
and the family on the left is summable if and only if the one on the right is.

(c) For every bounded operator $T$ on $H$ and every second Hilbert basis
$e' = (e'_j)_{j \in \kappa'}$ of $H$,
\[
  \sum_k \| T e_k \|^2 = \sum_j \| T^* e'_j \|^2 \qquad \text{in } [0, \infty] .
\]
\end{lemma}

\begin{proof}
\leanok
(a) Take real parts in Mathlib's Parseval identity
$\sum_k \langle x, e_k \rangle \langle e_k, x \rangle = \langle x, x \rangle$.

(c) Expanding $\|T e_k\|^2$ by (a) along $e'$ and $\|T^* e'_j\|^2$ along $e$ gives the
same double series $\sum_{j,k} |\langle e'_j, T e_k \rangle|^2$ with the two summations
exchanged. The rearrangement is carried out in $[0, \infty]$, where every sum is defined and
the interchange needs no hypothesis. For a self-adjoint $T$ it says that the
Hilbert--Schmidt sum does not depend on the basis.

(b) The case $e' = e$ of (c); summability is read off the value being finite.
\end{proof}

Part (b) is the hinge of the whole trace layer: it is what supplies the summability of
$P^{1/2} Q$ in \Cref{lem:operator-traces}(b), whose adjoint $Q P^{1/2}$ is visibly
Hilbert--Schmidt.

\begin{definition}[Trace of a positive operator]
\label{def:trace}
\lean{ContinuousLinearMap.traceAlong, ContinuousLinearMap.trace, HilbertBasis.chosen,
      ContinuousLinearMap.trace_eq_traceAlong, ContinuousLinearMap.traceAlong_eq_of_basis,
      ContinuousLinearMap.summable_re_inner_iff_of_basis,
      ContinuousLinearMap.summable_re_inner_chosen_iff,
      HilbertBasis.countable_of_separableSpace, HilbertBasis.nonempty_of_nontrivial,
      Discretization.Infinite.IsFiniteTracePos}
\leanok
\uses{lem:hilbert-schmidt-adjoint}
Let $H$ be a complex Hilbert space, let $T$ be a positive operator on $H$, and let
$e = (e_k)_{k \in \kappa}$ be a Hilbert basis of $H$. The \emph{trace} of $T$ is
\[
  \Tr T = \sum_{k \in \kappa} \re \langle e_k, T e_k \rangle \in [0, \infty] ,
\]
and $T$ is \emph{of finite trace} if $\Tr T < \infty$.
\end{definition}

The terms are nonnegative, because $\re \langle x, T x \rangle = \|T^{1/2} x\|^2$, so
$\Tr T = \sum_k \|T^{1/2} e_k\|^2$ is the squared Hilbert--Schmidt norm of $T^{1/2}$.
\Cref{lem:hilbert-schmidt-adjoint}(c) for the self-adjoint operator $T^{1/2}$ gives the same
value in $[0, \infty]$ along every Hilbert basis, so the trace does not depend on $e$. For
a positive operator $P$ of finite trace and a positive operator $Q$ write
$\Tr (P Q) = \sum_k \re \langle e_k, P Q e_k \rangle$; by \Cref{lem:operator-traces}(b) this
is the trace of the positive operator $P^{1/2} Q P^{1/2}$, so it does not depend on $e$
either.

In Lean the trace is a real number, \texttt{ContinuousLinearMap.trace}, computed along a
Hilbert basis chosen once and for all (\texttt{HilbertBasis.chosen}). The sum is taken to be
$0$ when the family is not summable; every statement assumes finite trace, so this
convention never enters. The same sum along a given basis $e$ is \texttt{traceAlong e}, and
for a positive operator the two agree (\texttt{ContinuousLinearMap.trace\_eq\_traceAlong}).
The chosen basis is countable when $H$ is separable and nonempty when $H$ is nonzero.
\texttt{IsFiniteTracePos J} bundles three properties of $J$: positive, of finite trace, and
injective.

\begin{lemma}[Traces of positive operators]
\label{lem:operator-traces}
\lean{ContinuousLinearMap.le_traceAlong_smul_one, ContinuousLinearMap.traceAlong_mul,
      ContinuousLinearMap.traceAlong_mul_pos, Discretization.Infinite.traceAlong_mul_le_of_le}
\leanok
\uses{def:trace}
Let $H$ be a complex Hilbert space.

(a) Every positive operator $T$ on $H$ of finite trace satisfies $T \preceq \Tr T \cdot 1$.

(b) Let $P$ be positive and of finite trace, let $Q$ be positive, and let
$e = (e_k)_k$ be a Hilbert basis of $H$. Then
\[
  \sum_k \re \langle e_k, P Q e_k \rangle
  = \sum_k \re \bigl\langle P^{1/2} e_k, \, Q \, P^{1/2} e_k \bigr\rangle
  \; \geq \; 0 ,
\]
and the value is positive as soon as $Q$ is invertible and $P \neq 0$.

(c) Let $P \preceq P'$ be positive and of finite trace, and let $Q$ be positive. Then
$\Tr (P Q) \leq \Tr (P' Q)$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:hilbert-schmidt-adjoint}
(a) For $x \in H$, writing $x$ in a Hilbert basis and applying the Cauchy--Schwarz
inequality to each term gives $\re \langle x, T x \rangle \leq \Tr T \cdot \|x\|^2$. This
is the operator analogue of \Cref{lem:loewner-basic-bounds}(b), and unlike the matrix proof
it uses no diagonalisation.

(b) The identity is the cyclicity $\Tr (PQ) = \Tr (P^{1/2} Q P^{1/2})$, read as a
symmetry of the pairing $\sum_k \langle S e_k, T e_k \rangle$ under passing to adjoints;
the required summability is \Cref{lem:hilbert-schmidt-adjoint}(b). Every term on the
right is nonnegative. For the strict inequality, invertibility of $Q$ makes $Q^{1/2}$
injective, so a vanishing trace would force $P^{1/2}$ to annihilate the whole basis.

(c) The difference $\Tr (P' Q) - \Tr (P Q)$ is the sum of (b) for the positive operator
$P' - P$ of finite trace, hence nonnegative.
\end{proof}

Invertibility of $Q$ here is what replaces positive definiteness. A positive operator of
finite trace on an infinite-dimensional space is compact, hence never bounded away from
zero, so positive definiteness is unavailable and the hypotheses on the two operators
have to be chosen differently: injectivity for $J$, and invertibility for the operator
the construction moves.

\begin{lemma}[Rank-one updates of operators]
\label{lem:operator-rank-one-updates}
\lean{ContinuousLinearMap.inverse_add_smul_rankOne,
      ContinuousLinearMap.rankOne_le_norm_sq_smul_one,
      Matrix.vecMulVec_le_norm_sq_smul_one}
\leanok
Let $H$ be a complex Hilbert space and $u \in H$, and write $u u^*$ for the rank-one
operator $z \mapsto \langle u, z \rangle u$.

(a) Let $A$ be a self-adjoint invertible operator on $H$ and $t$ a scalar with
$1 + t \langle u, A^{-1} u \rangle \neq 0$. Then $A + t \, u u^*$ is invertible and
\[
  \bigl( A + t \, u u^* \bigr)^{-1}
    = A^{-1} - \frac{t}{1 + t \langle u, A^{-1} u \rangle} \,
      \bigl( A^{-1} u \bigr) \bigl( A^{-1} u \bigr)^* .
\]

(b) $u u^* \preceq \|u\|^2 \cdot 1$.
\end{lemma}

\begin{proof}
\leanok
(a) The candidate is verified to be a two-sided inverse, exactly as in
\Cref{lem:rank-one-updates}(a). In a noncommutative ring one product does not suffice,
but two do: they exhibit $A + t \, u u^*$ as a unit.

(b) A self-adjoint element $a$ of a C$^*$-algebra satisfies $a \preceq \|a\| \cdot 1$,
and $u u^*$ is self-adjoint with $\|u u^*\| = \|u\|^2$. For $H = \mathbb{C}^{\kappa}$ the
formalisation also derives $v v^* \preceq \|v\|^2 \cdot 1$ from
\Cref{lem:loewner-basic-bounds}(b), since the trace of $v v^*$ is $\|v\|^2$.
\end{proof}

\section{The Gram operator}

\begin{lemma}[The Gram operator]
\label{lem:gram-operator}
\lean{ContinuousLinearMap.integrable_rankOne_self,
      ContinuousLinearMap.inner_integral_rankOne_self,
      ContinuousLinearMap.le_smul_one_iff_of_inner_eq_integral,
      ContinuousLinearMap.injective_iff_of_inner_eq_integral,
      ContinuousLinearMap.integral_rankOne_self_le_iff,
      ContinuousLinearMap.integral_rankOne_self_injective_iff,
      ContinuousLinearMap.nonneg_of_inner_eq_integral,
      ContinuousLinearMap.re_inner_eq_integral,
      ContinuousLinearMap.summable_re_inner_of_gram,
      ContinuousLinearMap.integral_re_inner_apply,
      Discretization.Infinite.inner_eq_integral_of_integral_rankOne,
      Discretization.Infinite.isFiniteTracePos_of_integral_rankOne,
      Discretization.Infinite.integral_norm_sq_eq_trace,
      Discretization.Infinite.integral_norm_sq_eq_traceAlong}
\leanok
\uses{def:trace}
Let $(\Omega, \mu)$ be a measure space, let $H$ be a complex Hilbert space, let
$b : \Omega \to H$ be square-integrable, and let $J = \int_\Omega b(x) b(x)^* \, d\mu(x)$.

(a) The operators $b(x) b(x)^*$ are Bochner integrable, and
$\langle u, J u \rangle = \int_\Omega |\langle u, b(x) \rangle|^2 \, d\mu(x)$ for every
$u \in H$.

(b) Let $\Lambda \in \mathbb{R}$. Then $J \preceq \Lambda \cdot 1$ if and only if
$\int_\Omega |\langle u, b(x) \rangle|^2 \, d\mu(x) \leq \Lambda \|u\|^2$ for every
$u \in H$.

(c) $J$ is injective if and only if every $u \in H$ with $\langle u, b(x) \rangle = 0$ for
almost every $x$ is zero.

(d) Let $H$ be separable. Then $J$ is positive and of finite trace, and
$\Tr J = \int_\Omega \|b(x)\|^2 \, d\mu(x)$.

(e) Let $H$ be separable and let $Q$ be a positive bounded operator on $H$. Then
\[
  \int_\Omega \re \bigl\langle b(x), Q \, b(x) \bigr\rangle \, d\mu(x) = \Tr (J Q).
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:hilbert-schmidt-adjoint, lem:operator-traces}
(a) The operator $b(x) b(x)^*$ has norm $\|b(x)\|^2$, which is integrable. Evaluation at
$u$ and the inner product with $u$ are continuous linear maps, so both pass under the
integral, and $\langle u, b(x) b(x)^* u \rangle = |\langle u, b(x) \rangle|^2$.

(b) By (a) both sides say that the quadratic form of the self-adjoint operator
$\Lambda \cdot 1 - J$ is nonnegative.

(c) The quadratic form vanishes at $u$ exactly when $\langle u, b(\cdot) \rangle$ vanishes
almost everywhere. If $J u = 0$ it vanishes. Conversely, $J$ is positive, and
$\langle u, J u \rangle = \|J^{1/2} u\|^2 = 0$ gives $J^{1/2} u = 0$ and hence $J u = 0$.

(d) Choose a Hilbert basis $(e_k)$ of $H$. Its index set is countable, because the open
balls of radius $1/2$ around the basis vectors are pairwise disjoint and a separable space
holds only countably many pairwise disjoint nonempty open sets. The quadratic form of (a) is
real and nonnegative, so $J$ is positive. By Tonelli and
\Cref{lem:hilbert-schmidt-adjoint}(a),
$\sum_k \langle e_k, J e_k \rangle = \sum_k \int_\Omega |\langle e_k, b \rangle|^2 \, d\mu
= \int_\Omega \|b\|^2 \, d\mu < \infty$, which is the finite trace and its value.

(e) Choose a countable Hilbert basis $(e_k)$ as in (d). Pointwise
$\re \langle b, Q b \rangle = \|Q^{1/2} b\|^2$, which
\Cref{lem:hilbert-schmidt-adjoint}(a) turns into
$\sum_k |\langle Q^{1/2} e_k, b \rangle|^2$. Sum and integral may be exchanged because all
terms are nonnegative and their integrals sum to a finite value, and the quadratic form
of $J$ turns the result into $\sum_k \re \langle Q^{1/2} e_k, J \, Q^{1/2} e_k \rangle$. That
equals $\Tr (J Q)$ because both are the squared Hilbert--Schmidt norm of
$Q^{1/2} J^{1/2}$, once read through the operator and once through its adjoint, by
\Cref{lem:hilbert-schmidt-adjoint}(b) and \Cref{lem:operator-traces}(b). Countability of
the index set of the basis enters here, in the interchange.
\end{proof}

\section{The upper potential and the verifier}

For the operator the construction moves, the right notion is positivity together with
invertibility; it is Mathlib's \texttt{IsStrictlyPositive}, and the inverse is
\texttt{Ring.inverse}, defined on all operators and equal to the inverse on the
invertible ones.

\begin{definition}[Upper potential and verifier of an operator]
\label{def:upper-potential}
\lean{Discretization.Infinite.upperPotential, Discretization.Infinite.upperVerifier,
      Discretization.upperPotential, Discretization.upperVerifier}
\leanok
\uses{def:trace}
Let $H$ be a complex Hilbert space, let $J$ be a positive injective operator on $H$ of
finite trace, let $B$ be a strictly positive operator on $H$, that is positive and
invertible, and let $\zeta > 0$. The \emph{upper potential} of $B$ is
$\Psi_J(B) = \Tr \bigl( J B^{-1} \bigr)$, and with $X = (B + \zeta \cdot J)^{-1}$ the
\emph{upper verifier} at $u \in H$ is
\[
  U_B^{\zeta}(u)
    = \frac{\re \langle u, X J X u \rangle}
           {\Psi_J(B) - \Psi_J(B + \zeta \cdot J)} + \re \langle u, X u \rangle .
\]
\end{definition}

The upper potential is large when $B$ is small in the directions where $J$ is large. Before
each step $B$ is grown by $\zeta \cdot J$, which lowers the potential by an amount that is
computed exactly, and the verifier decides whether a candidate point stays within that
budget, as $\Phi$ and $L_A^{\delta}$ do for the first family (\Cref{def:lower-potential}).
In Lean both notions are also defined directly for matrices
(\texttt{Discretization.upperPotential}, \texttt{Discretization.upperVerifier}). Lean
proves the lemmas below a second time for matrices, where the trace is the matrix trace,
and the read-off of \Cref{lem:upper-frame-bound}(b) once for every upper state; the matrix
proofs serve the finite case and keep the sparsification theorem elementary.

\begin{lemma}[How the upper potential changes]
\label{lem:upper-potential-change}
\lean{Discretization.Infinite.upperPotential_sub_eq,
      Discretization.Infinite.upperPotential_add_smul_lt,
      ContinuousLinearMap.isStrictlyPositive_add_smul,
      Discretization.Infinite.upperPotential_sub_smul_rankOne,
      Discretization.upperPotential_sub_eq, Discretization.upperPotential_add_smul_lt,
      Matrix.PosDef.add_smul, Discretization.upperPotential_sub_smul_vecMulVec}
\leanok
\uses{def:upper-potential}
Let $H$ be a nonzero complex Hilbert space, let $J$ be a positive injective operator on $H$
of finite trace, and let $B$ be a strictly positive operator on $H$.

(a) For every $\zeta \geq 0$ the operator $B + \zeta \cdot J$ is strictly positive and
\[
  \Psi_J(B) - \Psi_J(B + \zeta \cdot J)
    = \zeta \, \Tr \bigl( (J B^{-1} J) (B + \zeta \cdot J)^{-1} \bigr),
\]
and for $\zeta > 0$ the right-hand side is positive.

(b) Let $C$ be a strictly positive operator on $H$, let $u \in H$, and let $w \geq 0$ with
$w \re \langle u, C^{-1} u \rangle < 1$. Then
\[
  \Psi_J \bigl( C - w \, u u^* \bigr)
    = \Psi_J(C)
      + \frac{w}{1 - w \re \langle u, C^{-1} u \rangle} \,
        \re \langle u, C^{-1} J C^{-1} u \rangle .
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:operator-traces, lem:operator-rank-one-updates}
(a) The operator $B + \zeta \cdot J$ dominates $B$, so it is positive, and invertible
because it dominates an invertible positive operator. The identity is the resolvent
identity, which holds in any ring. The right-hand side is the trace of a product of two
positive operators, the first of them $J B^{-1} J$, which is again of finite trace and
positive; \Cref{lem:operator-traces}(b) gives the sign, and injectivity of $J$ makes it
strict.

(b) Insert the Sherman--Morrison formula of \Cref{lem:operator-rank-one-updates}(a) with
$t = -w$ and take traces, as for the lower potential in
\Cref{lem:lower-potential-change}(b).
\end{proof}

\begin{lemma}[Barrier lemma for the upper potential]
\label{lem:upper-barrier}
\lean{Discretization.Infinite.upperPotential_update_le,
      Discretization.upperPotential_update_le}
\leanok
\uses{def:upper-potential}
Let $H$ be a nonzero complex Hilbert space, let $J$ be a positive injective operator on $H$
of finite trace, let $B$ be a strictly positive operator on $H$, let $\zeta > 0$,
$u \in H$, and let $w > 0$ satisfy $U_B^{\zeta}(u) \leq 1/w$. Then
$B + \zeta \cdot J - w \, u u^*$ is strictly positive and
\[
  \Psi_J \bigl( B + \zeta \cdot J - w \, u u^* \bigr) \leq \Psi_J(B).
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:upper-potential-change, lem:operator-rank-one-updates}
Write $X = (B + \zeta \cdot J)^{-1}$. The first summand of $U_B^{\zeta}(u)$ is positive
for $u \neq 0$, because $\re \langle u, X J X u \rangle = \|J^{1/2} X u\|^2$ and $J$ is
injective, so the hypothesis gives $w \re \langle u, X u \rangle < 1$: the
Sherman--Morrison denominator is positive. The inverse of
\Cref{lem:operator-rank-one-updates}(a) is then positive, which makes
$B + \zeta \cdot J - w \, u u^*$ strictly positive. Finally
\Cref{lem:upper-potential-change}(b) for $C = B + \zeta \cdot J$ and (a) reduce the
inequality to the hypothesis $U_B^{\zeta}(u) \leq 1/w$ multiplied out, as in the proof of
\Cref{lem:lower-barrier}.
\end{proof}

\section{The upper verifier passes on average}

For a square-integrable $b$ with values in a separable space, the Gram operator
$J = \int_\Omega b(x) b(x)^* \, d\mu(x)$ is positive and of finite trace
(\Cref{lem:gram-operator}(d)), so its upper potential is defined.

\begin{lemma}[The upper verifier passes on average]
\label{lem:upper-verifier-on-average}
\lean{Discretization.Infinite.integrable_upperVerifier,
      Discretization.Infinite.integral_upperVerifier_lt,
      Discretization.integrable_upperVerifier, Discretization.integral_upperVerifier_lt}
\leanok
\uses{def:upper-potential}
Let $(\Omega, \mu)$ be a measure space, let $H$ be a nonzero separable complex Hilbert
space, let $b : \Omega \to H$ be square-integrable with injective Gram operator
$J = \int_\Omega b(x) b(x)^* \, d\mu(x)$, let $B$ be a strictly positive operator on $H$,
and let $\zeta > 0$. Then $x \mapsto U_B^{\zeta}(b(x))$ is integrable and
\[
  \int_\Omega U_B^{\zeta} \bigl( b(x) \bigr) d\mu(x) < \frac{1}{\zeta} + \Psi_J(B).
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:gram-operator, lem:operator-traces, lem:upper-potential-change,
      lem:loewner-basic-bounds}
Both summands of the verifier are quadratic forms in $b(x)$, bounded by a multiple of
$\|b(x)\|^2$, so the verifier is integrable. Write $W = B^{-1}$ and
$X = (B + \zeta \cdot J)^{-1}$, and let $E = \Psi_J(B) - \Psi_J(B + \zeta \cdot J)$ be the
gain of \Cref{lem:upper-potential-change}(a). By \Cref{lem:gram-operator}(e) for
$Q = X J X$ and $Q = X$ the average equals
$\Tr (J X J X) / E + \Psi_J(B + \zeta \cdot J)$. Since $X \preceq W$ by the antitonicity of
the inverse, as in \Cref{lem:loewner-basic-bounds}(c), we have $J X J \preceq J W J$, and
\Cref{lem:operator-traces}(c) bounds the numerator by $\Tr (J W J X) = E / \zeta$. So the
first summand is at most $1/\zeta$, and the second is strictly smaller than $\Psi_J(B)$
by \Cref{lem:upper-potential-change}(a).
\end{proof}

\section{From the upper potential to the frame bound}

\begin{lemma}[From the potential to the upper frame bound]
\label{lem:upper-frame-bound}
\lean{Discretization.Infinite.inv_upperPotential_smul_le,
      Matrix.PosDef.inv_re_trace_mul_smul_le,
      Discretization.UpperBarrier.upper_bound_of_state,
      Discretization.UpperBarrier.upper_frame_bound}
\leanok
\uses{def:upper-potential}
Let $H$ be a nonzero complex Hilbert space, let $J$ be a positive injective operator on $H$
of finite trace, and let $B$ be a strictly positive operator on $H$.

(a) $\Psi_J(B)^{-1} \cdot J \preceq B$.

(b) Let $\Lambda > 0$ with $J \preceq \Lambda \cdot 1$, let $n \geq 1$, let $s \geq 1/n$
with $\Tr J = \Lambda (n s^2 + 1)$, put $\zeta = (1 + s)/n$ and $d_0 = \zeta \Tr J / s$, let
$u_1, \dots, u_n \in H$ and $w_1, \dots, w_n \in \mathbb{R}$, and let
$B_n = d_0 \cdot 1 + n \zeta \cdot J - \sum_{i=1}^n w_i \, u_i u_i^*$. If $B_n$ is
strictly positive and $\Psi_J(B_n) \leq s / \zeta$, then
\[
  \sum_{i=1}^n w_i \, u_i u_i^* \; \preceq \; (1 + s)^2 \Lambda \cdot 1 .
\]
\end{lemma}

\begin{proof}
\leanok
\uses{lem:operator-traces, lem:sqrt-conjugation}
(a) The operator $B^{-1/2} J B^{-1/2}$ is positive and of finite trace, and its trace is
again $\Psi_J(B)$ by \Cref{lem:operator-traces}(b), which is positive since $J \neq 0$ on
the nonzero space $H$. So \Cref{lem:operator-traces}(a) bounds it by $\Psi_J(B) \cdot 1$,
and \Cref{lem:sqrt-conjugation} carries that back to $J \preceq \Psi_J(B) \cdot B$.

(b) By (a), $\Psi_J(B_n) \leq s/\zeta$ gives $(\zeta / s) \cdot J \preceq B_n$, hence
$\sum_i w_i \, u_i u_i^* \preceq d_0 \cdot 1 + (n\zeta - \zeta/s) \cdot J$. The
coefficient $n\zeta - \zeta/s = \zeta (n - 1/s)$ is nonnegative exactly because
$s \geq 1/n$, so $J \preceq \Lambda \cdot 1$ may be applied to it, and
$d_0 + (n\zeta - \zeta/s)\Lambda = \zeta \Lambda (n s + n) = (1 + s)^2 \Lambda$. In Lean
this read-off is proved once for every upper state,
\texttt{Discretization.UpperBarrier.upper\_bound\_of\_state}; the arithmetic is in
\texttt{Discretization/Parameters.lean}.
\end{proof}

For $s = \sqrt{(M-1)/n}$ one has $n s^2 + 1 = M$, and $s \geq 1/n$ is the condition
$M \geq 1 + 1/n$; the lemma is stated in $s$ because its proof never unfolds the square
root.

\chapter{The construction and the proof}
\label{ch:construction}

The construction carries a matrix over $\iota$ for the first family and an operator on $H$
for the second, each controlled by its potential. This chapter runs it and proves
\Cref{thm:discretization}.

\section{The barrier iteration}

\begin{proposition}[The barrier iteration]
\label{prop:barrier-iteration}
\lean{Discretization.lowerState, Discretization.UpperBarrier.upperState,
      Discretization.UpperBarrier.exists_points_weights, Discretization.UpperBarrier,
      Discretization.matrixUpperBarrier, Discretization.Infinite.operatorUpperBarrier}
\leanok
\uses{def:lower-potential, def:upper-potential}
Let $(\Omega, \mu)$ be a measure space, let $\iota$ be a finite nonempty set, and let $H$
be a nonzero separable complex Hilbert space. Let $a : \Omega \to \mathbb{C}^{\iota}$ be
square-integrable in every coordinate with $\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$, and
let $b : \Omega \to H$ be square-integrable with injective Gram operator
$J = \int_\Omega b(x) b(x)^* \, d\mu(x)$. Let $A_0 \in \mathbb{C}^{\iota \times \iota}$ be
positive definite, let $B_0$ be a strictly positive operator on $H$, and let
$\delta, \zeta > 0$ satisfy the initial gap condition
\[
  \frac{1}{\zeta} + \Psi_J(B_0) \; \leq \; \frac{1}{\delta} - \Phi(A_0).
\]
For points $x_1, \dots, x_k \in \Omega$ and weights $w_1, \dots, w_k$ write
\[
  A_k = A_0 - k \delta \cdot 1 + \sum_{i=1}^k w_i \, a(x_i) a(x_i)^*,
  \qquad
  B_k = B_0 + k \zeta \cdot J - \sum_{i=1}^k w_i \, b(x_i) b(x_i)^* .
\]
Then for every $k \geq 0$ there are points $x_1, \dots, x_k \in \Omega$ and weights
$w_1, \dots, w_k > 0$ such that $A_k$ is positive definite, $B_k$ is strictly positive,
$\Phi(A_k) \leq \Phi(A_0)$ and $\Psi_J(B_k) \leq \Psi_J(B_0)$.
\end{proposition}

\begin{proof}
\leanok
\uses{lem:lower-verifier-on-average, lem:upper-verifier-on-average, lem:lower-barrier,
      lem:upper-barrier}
Induction on $k$; for $k = 0$ there is nothing to show. The invariant keeps the gap open,
$1/\zeta + \Psi_J(B_k) \leq 1/\zeta + \Psi_J(B_0) \leq 1/\delta - \Phi(A_0)
\leq 1/\delta - \Phi(A_k)$, and in particular $\delta < \Phi(A_k)^{-1}$. By
\Cref{lem:upper-verifier-on-average} the function $U(x) = U_{B_k}^{\zeta}(b(x))$ is
integrable with average below $1/\zeta + \Psi_J(B_k)$, so
\Cref{lem:lower-verifier-on-average}(b) produces a point $x_{k+1}$ with
$U(x_{k+1}) < L_{A_k}^{\delta}(a(x_{k+1}))$. The weight is
$w_{k+1} = 1 / L_{A_k}^{\delta}(a(x_{k+1}))$, positive because $U \geq 0$, and
\Cref{lem:lower-barrier} and \Cref{lem:upper-barrier} restore the invariant.
\end{proof}

The number of steps is arbitrary; only the initial gap condition drives it. What ties the
number of points to the frame bounds is the choice of the initial states and of the two
shifts, made in the proof of \Cref{thm:discretization} below.

The induction and the read-off use only a few properties of the upper state: that
$\Psi_J(B)$ and $U_B^{\zeta}$ are positive, \Cref{lem:upper-barrier} and
\Cref{lem:upper-verifier-on-average}, the bound $\Psi_J(B)^{-1} \cdot J \preceq B$ with the
potential of a multiple of the identity, and the crude bound
$b(y) b(y)^* \preceq \|b(y)\|^2 \cdot 1$ with $\int_\Omega \|b\|^2 \, d\mu = \Tr J$. In
Lean they are collected in \texttt{Discretization.UpperBarrier}, and the construction, the
read-off of the upper frame bound and the theorem with its edge cases are proved once for
every such upper state. The operators of \Cref{ch:infinite} are one instance
(\texttt{Discretization.Infinite.operatorUpperBarrier}); the matrices over a finite set
$\kappa$, with the matrix proofs of the upper lemmas, are another
(\texttt{Discretization.matrixUpperBarrier}).

\section{Proof of the discretization theorem}

\begin{proof}[Proof of \Cref{thm:discretization}]
\proves{thm:discretization}
\leanok
\uses{lem:gram-identities, lem:gram-operator, prop:barrier-iteration,
      lem:lower-frame-bound, lem:upper-frame-bound, lem:lower-barrier, lem:upper-barrier,
      lem:lower-verifier-on-average, lem:upper-verifier-on-average,
      lem:point-from-averages, lem:operator-rank-one-updates}
The first family is normalized first. Replace $a$ by $I^{-1/2} a$, whose Gram matrix is
$I^{-1/2} I \, I^{-1/2} = 1$ by \Cref{lem:gram-identities}(b). Once points and weights are
found for it, conjugating the lower bound by $I^{1/2}$ preserves the Loewner order and
turns $(1 - r)^2 \cdot 1$, with $r = \sqrt{(m-1)/n}$, into $(1 - r)^2 \cdot I$; the upper
bound does not involve the first family. From now on
$\int_\Omega a(x) a(x)^* \, d\mu(x) = 1$.

By \Cref{lem:gram-operator}(d), $J$ is positive and of finite trace with
$\Tr J = \int_\Omega \|b\|^2 \, d\mu = M \Lambda$, so the upper potential and verifier of
\Cref{def:upper-potential} are defined. Put $r = \sqrt{(m-1)/n}$, so that
$n r^2 + 1 = m$, and for $M \geq 1$ put $s = \sqrt{(M-1)/n}$, so that $n s^2 + 1 = M$.

Case $m \geq 2$, $M \geq 1 + 1/n$ (\texttt{Discretization.bss\_generalized\_of\_gram\_eq\_one}),
the potential argument. Put
\[
  \delta = \tfrac{1-r}{n}, \quad \zeta = \tfrac{1+s}{n}, \quad
  A_0 = \tfrac{\delta m}{r} \cdot 1, \quad B_0 = \tfrac{\zeta \Tr J}{s} \cdot 1 .
\]
Since $2 \leq m \leq n$ we have $0 < r < 1$, and $M \geq 1 + 1/n$ gives $s \geq 1/n$.
Then $\Phi(A_0) = r/\delta$ and $\Psi_J(B_0) = s/\zeta$, so the initial gap closes
exactly, $1/\delta - \Phi(A_0) = n = 1/\zeta + \Psi_J(B_0)$.
\Cref{prop:barrier-iteration} with $k = n$ gives points and weights with
$\Phi(A_n) \leq r/\delta$ and $\Psi_J(B_n) \leq s/\zeta$, and
\Cref{lem:lower-frame-bound} and \Cref{lem:upper-frame-bound}(b) turn them into
the two bounds. The remaining three cases need one potential only: the verifier on the
other side is replaced by a constant, whose average is again $n$. For $m = 1$ the lower
constant $(1 - r)^2$ is $1$.

Case $m = 1$, $M \geq 1 + 1/n$ (\texttt{Discretization.bss\_generalized\_of\_unique}). The
lower verifier is replaced by $L(x) = n |a(x)|^2$, whose average is $n$ because the single
function is normalized. The induction then tracks only the upper state, using
\Cref{lem:upper-barrier} and the average of \Cref{lem:upper-verifier-on-average} with
\Cref{lem:point-from-averages}(a), and takes the largest weight the constant allows,
$1/w_i = L(x_i)$. Each point therefore contributes exactly $1/n$ to the lower bound, which
becomes the identity $\sum_i w_i |a(x_i)|^2 = 1$. The upper bound is
\Cref{lem:upper-frame-bound}(b).

Case $m \geq 2$, $M \leq 1 + 1/n$
(\texttt{Discretization.bss\_generalized\_of\_small\_dim}). The upper verifier is replaced
by $U(x) = n \|b(x)\|^2 / \Tr J$, whose average is $n$. The induction tracks only the lower
matrix, with \Cref{lem:lower-barrier}, and the lower bound is
\Cref{lem:lower-frame-bound}. The weights it produces satisfy
$w_i U(x_i) \leq 1$, that is $w_i \|b(x_i)\|^2 \leq \Tr J / n$. Summing over the $n$
points and using \Cref{lem:operator-rank-one-updates}(b) gives
$\sum_i w_i \, b(x_i) b(x_i)^* \preceq \Tr J \cdot 1 = M \Lambda \cdot 1$, and
$M \leq 1 + 1/n$ forces $s = \sqrt{(M-1)/n} \leq 1/n$ and hence
$M = 1 + n s^2 \leq 1 + s \leq (1+s)^2$.

Case $m = 1$, $M \leq 1 + 1/n$
(\texttt{Discretization.bss\_generalized\_of\_unique\_of\_small\_dim}). Both verifiers are
now the constants $L(x) = n |a(x)|^2$ and $U(x) = n \|b(x)\|^2 / \Tr J$, of equal average
$n$, and neither depends on the state of the construction.
\Cref{lem:point-from-averages}(b) produces a single point $y$ with $U(y) \leq L(y)$ and
$L(y) > 0$; using it $n$ times with the weight $w = 1/L(y)$ makes the lower bound the
identity $\sum_i w_i |a(x_i)|^2 = 1$, while $w \, U(y) \leq 1$ and
\Cref{lem:operator-rank-one-updates}(b) give the upper bound as in the previous case. No
induction is needed.
\end{proof}

In Lean the normalisation is \texttt{Discretization.bss\_of\_gram\_eq\_one}, stated once
for an arbitrary conclusion about the second family, and the four cases are proved once
for every upper state of \texttt{Discretization.UpperBarrier}. A finite second family,
$H = \mathbb{C}^{\kappa}$, is proved with the matrix instance, which keeps \Cref{cor:bss}
free of operators.

\section{A countable family}

\begin{lemma}[A countable family as a map into $\ell_2$]
\label{lem:countable-gram-operator}
\lean{Discretization.Countable.ae_mem_l2, Discretization.Countable.exists_measurableSet_mem_l2,
      Discretization.Countable.toLp, Discretization.Countable.memLp_toLp,
      Discretization.Countable.inner_toLp, Discretization.Countable.integral_norm_sq_toLp,
      Discretization.Countable.gramOp, Discretization.Countable.gramOp_injective,
      Discretization.Countable.gramOp_le, Discretization.Countable.integral_comap_val,
      Discretization.Countable.gram_comap_val, HilbertBasis.separableSpace}
\leanok
Let $(\Omega, \mu)$ be a measure space, let $\kappa$ be a countable set, and let
$b = (b_k)_{k \in \kappa}$ be square-integrable functions on $\Omega$ with
$\sum_k \|b_k\|_{L_2}^2 < \infty$.

(a) There is a measurable set $S \subseteq \Omega$ of full measure with
$\sum_k |b_k(x)|^2 < \infty$ for every $x \in S$.

(b) If $\sum_k |b_k(x)|^2 < \infty$ for every $x \in \Omega$, then
$x \mapsto b(x) = (b_k(x))_k$ is a square-integrable map into the separable space
$\ell_2(\kappa)$ with $\int_\Omega \|b(x)\|^2 \, d\mu(x) = \sum_k \|b_k\|_{L_2}^2$, and
$\langle c, b(x) \rangle = \sum_k \overline{c_k} \, b_k(x)$ for every $c \in \ell_2(\kappa)$
and every $x \in \Omega$.

(c) Under the hypothesis of (b), the Gram operator $J = \int_\Omega b(x) b(x)^* \, d\mu(x)$
satisfies $J \preceq \Lambda \cdot 1$ if
$\int_\Omega |\sum_k \overline{c_k} \, b_k(x)|^2 \, d\mu(x) \leq \Lambda \|c\|^2$ for every
$c \in \ell_2(\kappa)$, and it is injective if this integral vanishes only for $c = 0$.
\end{lemma}

\begin{proof}
\leanok
\uses{lem:gram-operator}
(a) By Tonelli, $\int_\Omega \sum_k |b_k|^2 \, d\mu = \sum_k \|b_k\|^2 < \infty$, so the
sum is finite almost everywhere, and the complement of a measurable null set containing
the exceptional points is a set $S$ as required.

(b) The finite partial sums $\sum_{k \in F} b_k(x) e_k$ converge to $b(x)$ in
$\ell_2(\kappa)$, so $b$ is strongly measurable, and $\int_\Omega \|b\|^2 \, d\mu =
\sum_k \|b_k\|^2$ by Tonelli. The inner product of $\ell_2(\kappa)$ is
$\langle c, v \rangle = \sum_k \overline{c_k} \, v_k$. The space is separable because its
standard basis $(e_k)$ is countable: the span of a countable set is separable, and the span
of a Hilbert basis is dense.

(c) By (b) the integral is $\int_\Omega |\langle c, b(x) \rangle|^2 \, d\mu(x)$, so the two
claims are \Cref{lem:gram-operator}(b) and (c); the integral vanishes exactly when
$\langle c, b(\cdot) \rangle = 0$ almost everywhere.
\end{proof}

Restricting the measure space to the set $S$ of (a) changes no integral and no Gram
matrix, because $S$ has full measure, and it makes (b) applicable. This is how the proof of
\Cref{cor:countable-discretization} chooses its points inside $S$: the construction runs in
$S$.

\section{A reproducing kernel Hilbert space}

\begin{lemma}[Kernel sections]
\label{lem:kernel-sections}
\lean{Discretization.RKHS.evalCLM, Discretization.RKHS.kernelSection,
      Discretization.RKHS.inner_kernelSection, Discretization.RKHS.norm_inner_kernelSection,
      Discretization.RKHS.kernel, Discretization.RKHS.kernel_eq,
      Discretization.RKHS.kernel_self,
      Discretization.RKHS.aestronglyMeasurable_kernelSection,
      Discretization.RKHS.memLp_kernelSection}
\leanok
Let $H$ be a complex Hilbert space of functions on a set $\Omega$ on which every point
evaluation $f \mapsto f(x)$ is continuous.

(a) For every $x \in \Omega$ there is $k_x \in H$ with $\langle k_x, f \rangle = f(x)$ for
every $f \in H$. With $K(x, y) = \langle k_y, k_x \rangle$ one has $K(x, y) = k_x(y)$, so
$k_x = K(x, \cdot)$ and $K(x, x) = \|k_x\|^2$, and $|\langle f, k_x \rangle| = |f(x)|$ for
every $f \in H$.

(b) Let $(\Omega, \mu)$ be a measure space, let $H$ be separable, and let every $f \in H$
be measurable. Then $x \mapsto k_x$ is strongly measurable, and it is square-integrable if
$\int_\Omega K(x, x) \, d\mu(x) < \infty$.
\end{lemma}

\begin{proof}
\leanok
(a) Evaluation at $x$ is a continuous linear functional, so the Riesz representation
theorem gives $k_x$. Then $K(x, y) = \langle k_y, k_x \rangle = k_x(y)$,
$K(x, x) = \|k_x\|^2$, and $\langle f, k_x \rangle = \overline{\langle k_x, f \rangle}
= \overline{f(x)}$.

(b) Take a Hilbert basis $(e_j)$ of $H$, which is countable because $H$ is separable. Then
$k_x = \sum_j \langle e_j, k_x \rangle \, e_j$ with
$\langle e_j, k_x \rangle = \overline{e_j(x)}$, so $x \mapsto k_x$ is the pointwise limit
of finite sums of measurable maps. Its squared norm is $K(x, x)$.
\end{proof}

In Lean a Hilbert space of functions is a Hilbert space $H$ together with a linear map
$\varphi : H \to \mathbb{C}^{\Omega}$, $\varphi(f)(x)$ being the value of $f$ at $x$.
Injectivity of $\varphi$ is not required; in \Cref{cor:rkhs-discretization} it follows from
the assumption on vanishing functions.

\chapter{Discretizing the uniform norm}
\label{ch:design}

The discretization theorem starts from a measure; this chapter produces one. Let $D$ be
an arbitrary set, $\iota$ a finite nonempty index set with $m = \#\iota$ elements, and
$a = (a_k)_{k \in \iota}$ a family of bounded functions $D \to \mathbb{C}$ whose
coordinate functions are linearly independent. For every $\varepsilon > 0$ we look for
finitely many points $x_k \in D$ with weights $w_k \geq 0$ summing to one for which
\[
  |f(y)|^2 \; \leq \; (m + \varepsilon) \sum_k w_k |f(x_k)|^2
  \qquad \text{for all } y \in D \text{ and all } f \in \operatorname{span}\{a_k\},
\]
that is, the uniform norm is dominated by the $L_2$ norm of that measure with the
constant $\sqrt{m + \varepsilon}$. The answer of Kiefer and Wolfowitz~\cite{KW} is that a
measure whose Gram matrix has an almost maximal determinant does this. The complex and
non-compact case is treated in~\cite{KPUU}.

No topology and no measurable structure on $D$ are used: a finitely supported measure is
a finite list of points with weights, and it is kept in that form until
\Cref{cor:design-as-measure}.

A \emph{design} consists of finitely many points $x_1, \dots, x_N \in D$ and weights
$w_1, \dots, w_N \geq 0$ with $\sum_k w_k = 1$, and its \emph{Gram matrix} is
$G = \sum_k w_k \, a(x_k) a(x_k)^*$, the Gram matrix of $a$ in $L_2$ of the measure
$\sum_k w_k \delta_{x_k}$; in Lean it is
\texttt{Discretization.KieferWolfowitz.designGram}. The statements below spell both out.

\begin{lemma}[Quadratic form of the Gram matrix of a design]
\label{lem:design-quadratic-form}
\lean{Discretization.KieferWolfowitz.designGram,
      Discretization.KieferWolfowitz.re_quadForm_designGram}
\leanok
Let $D$ be a set, let $\iota$ be a finite set and let $a : D \to \mathbb{C}^{\iota}$. For
points $x_1, \dots, x_N \in D$, weights $w_1, \dots, w_N \geq 0$ with $\sum_k w_k = 1$
and $G = \sum_k w_k \, a(x_k) a(x_k)^*$, and for every $c \in \mathbb{C}^{\iota}$, one has
$c^* G c = \sum_k w_k |\langle c, a(x_k) \rangle|^2$.
\end{lemma}

\begin{proof}
\leanok
The quadratic form of a rank-one matrix $u u^*$ at $c$ is $|\langle c, u \rangle|^2$,
and both sides are additive in the summands.
\end{proof}

\begin{lemma}[Gain from mixing in one point]
\label{lem:mixing-gain}
\lean{Discretization.KieferWolfowitz.exists_mix_ge}
\leanok
Let $m \geq 1$ and $t > m$ be real. Then $\alpha = (t-m) / (2m(t-1))$ lies in $[0,1)$ and
\[
  1 + \frac{(t-m)^2}{4m(t-1)}
    \; \leq \; (1-\alpha)^{m-1} \bigl( 1 + \alpha (t-1) \bigr).
\]
\end{lemma}

\begin{proof}
\leanok
By Bernoulli's inequality $(1-\alpha)^{m-1} \geq 1 - (m-1)\alpha$, and the difference of
the two sides of the claim is then the explicit nonnegative quantity
$(t-m)^2 / (4m^2(t-1))$. The weight $\alpha$ is the one for which the linear gain
$\alpha(t-m)$ is twice the quadratic loss $(m-1)(t-1)\alpha^2$, so half of the gain
survives.
\end{proof}

\begin{proposition}[Almost maximal determinant]
\label{prop:almost-maximal-determinant}
\lean{Discretization.KieferWolfowitz.exists_design_quadForm_inv_le,
      Discretization.KieferWolfowitz.exists_design_posDef}
\leanok
Let $\iota$ be a finite nonempty index set with $m = \#\iota$ elements, let
$a = (a_k)_{k \in \iota}$ be bounded functions on a set $D$ whose coordinate functions
are linearly independent, and let $\varepsilon > 0$. Then there are points
$x_1, \dots, x_N \in D$ and weights $w_1, \dots, w_N \geq 0$ with $\sum_k w_k = 1$ such
that $G = \sum_k w_k \, a(x_k) a(x_k)^*$ is positive definite and
\[
  a(y)^* G^{-1} a(y) \; \leq \; m + \varepsilon \qquad \text{for all } y \in D .
\]
\end{proposition}

\begin{proof}
\leanok
\uses{lem:design-quadratic-form, lem:rank-one-updates, lem:mixing-gain}
First, some design has a positive definite Gram matrix
(\texttt{Discretization.KieferWolfowitz.exists\_design\_posDef}). Linear independence of
the coordinate functions says that $c = 0$ is the only coefficient vector with
$\langle c, a(y) \rangle = 0$ for all $y \in D$. It therefore suffices to find points
annihilating no nonzero coefficient vector, for then the uniform distribution on them has
$c^* G c > 0$ for every $c \neq 0$ by \Cref{lem:design-quadratic-form}. Descending
induction on dimension supplies them: while some nonzero vector survives the points
chosen so far, the hypothesis gives a point at which it does not vanish, and adding that
point lowers the dimension.

The determinants of the Gram matrices of designs form a set of real numbers which is
bounded above, since the entries of every Gram matrix are bounded by $C^2$ and the
Leibniz formula bounds a determinant by its entries, and which contains a positive
number by the first step. Let $S$ be its supremum and take a design whose
determinant exceeds $S / (1 + \eta)$, where
$\eta = \varepsilon^2 / (4m(m + \varepsilon - 1))$; its Gram matrix $G$ is positive
definite because its determinant is positive.

Appending a point $y$ with weight $\alpha \in [0,1)$ and scaling the old weights by
$1 - \alpha$ is again a design, with Gram matrix $(1-\alpha) G + \alpha \, a(y) a(y)^*$.
By \Cref{lem:rank-one-updates}(c) its determinant is $\det G$ multiplied by
$(1-\alpha)^{m-1}(1 + \alpha(t-1))$ with $t = a(y)^* G^{-1} a(y)$, and that multiple
cannot exceed $S / \det G < 1 + \eta$. If $t$ were larger than $m + \varepsilon$, the
weight of \Cref{lem:mixing-gain} would make it at least $1 + \eta$.
\end{proof}

\begin{theorem}[Discretization of the uniform norm]
\label{thm:uniform-discretization}
\lean{Discretization.KieferWolfowitz.exists_design_kieferWolfowitz,
      Discretization.KieferWolfowitz.exists_design_kieferWolfowitz_card_le}
\leanok
Let $\iota$ be a finite nonempty index set with $m = \#\iota$ elements, let
$a = (a_k)_{k \in \iota}$ be bounded functions on a set $D$ with linearly independent
coordinate functions, and let $\varepsilon > 0$. Then there are points
$x_1, \dots, x_n \in D$ and weights $w_1, \dots, w_n \geq 0$ with $\sum_k w_k = 1$ such
that
\[
  |f(y)|^2 \; \leq \; (m + \varepsilon) \sum_{k=1}^n w_k \, |f(x_k)|^2
\]
for every $y \in D$ and every $f \in \operatorname{span}\{a_k\}$. The points can be taken
to number at most $2m^2 + 1$.
\end{theorem}

\begin{proof}
\leanok
\uses{prop:almost-maximal-determinant, lem:design-quadratic-form, lem:rank-one-updates,
      lem:few-design-points}
Take the design of \Cref{prop:almost-maximal-determinant} for $\varepsilon/2$. For $s = m + \varepsilon$
the bound $s^{-1} a(y)^* G^{-1} a(y) < 1$ is strict, so subtracting $s^{-1} a(y) a(y)^*$
from $G$ leaves a positive definite matrix by \Cref{lem:rank-one-updates}(b), that is
$a(y) a(y)^* \preceq s \cdot G$. Evaluating this at the coefficient vector of $f$ and
using \Cref{lem:design-quadratic-form} gives the claim. The inequality depends on the design
only through its Gram matrix, which \Cref{lem:few-design-points} preserves while cutting the
design down to $2m^2 + 1$ points.
\end{proof}

\begin{theorem}[The sharp constant on a compact domain]
\label{thm:uniform-discretization-compact}
\lean{Discretization.KieferWolfowitz.exists_design_kieferWolfowitz_of_compact,
      Discretization.KieferWolfowitz.exists_design_kieferWolfowitz_of_compact_card_le,
      IsCompact.convexHull}
\leanok
Let $\iota$ be a finite nonempty index set with $m = \#\iota$ elements, let $D$ be a
compact space and let $a = (a_k)_{k \in \iota}$ be continuous functions on it with
linearly independent coordinate functions. Then there are points
$x_1, \dots, x_n \in D$ and weights $w_1, \dots, w_n \geq 0$ with $\sum_k w_k = 1$ such
that
\[
  |f(y)|^2 \; \leq \; m \sum_{k=1}^n w_k \, |f(x_k)|^2
\]
for every $y \in D$ and every $f \in \operatorname{span}\{a_k\}$. The points can be taken
to number at most $2m^2 + 1$.
\end{theorem}

\begin{proof}
\leanok
\uses{prop:almost-maximal-determinant, lem:mixing-gain, lem:design-quadratic-form,
      lem:few-design-points}
The matrices $a(y) a(y)^*$ form a compact set, whose convex hull is compact as well and
consists exactly of the Gram matrices of designs. The determinant is continuous, so it
attains its maximum at a design. No mixture can then increase the determinant at all,
and \Cref{lem:mixing-gain} applied with the factor $1$ gives $a(y)^* G^{-1} a(y) \leq m$ without
slack. The passage to the norm inequality is that of
\Cref{thm:uniform-discretization} for every constant $s > m$, followed by the limit
$s \to m$, which is a limit of real numbers, and the point count is again
\Cref{lem:few-design-points}.
\end{proof}

Both theorems produce a list of points with weights. Read as a measure, that list is
the object the discretization theorem asks for.

\begin{corollary}[The design as a measure]
\label{cor:design-as-measure}
\lean{Discretization.KieferWolfowitz.exists_probabilityMeasure_kieferWolfowitz}
\leanok
Let $\iota$ be a finite nonempty index set with $m = \#\iota$ elements, let $D$ carry a
measurable structure, let $a = (a_k)_{k \in \iota}$ be bounded measurable functions on it
with linearly independent coordinate functions, and let $\varepsilon > 0$. Then there is
a finitely supported probability measure $\varrho = \sum_k w_k \delta_{x_k}$ on $D$ whose
Gram matrix $\int_D a(y) a(y)^* \, d\varrho(y)$ is positive definite and equal to
$\sum_k w_k \, a(x_k) a(x_k)^*$, and
\[
  |f(y)|^2 \; \leq \; (m + \varepsilon) \int_D |f|^2 \, d\varrho
\]
for every $y \in D$ and every $f \in \operatorname{span}\{a_k\}$.
\end{corollary}

\begin{proof}
\leanok
\uses{thm:uniform-discretization}
This is \Cref{thm:uniform-discretization} read through the measure: integration against $\sum_k w_k \delta_{x_k}$ is the weighted sum over the points, for
every strongly measurable integrand.
\end{proof}

\begin{lemma}[Few points suffice]
\label{lem:few-design-points}
\lean{Discretization.KieferWolfowitz.exists_design_card_le,
      Discretization.KieferWolfowitz.designGram_mem_convexHull,
      Discretization.KieferWolfowitz.exists_design_of_mem_convexHull}
\leanok
Let $D$ be a set, let $\iota$ be a finite set with $m = \#\iota$ elements and let
$a : D \to \mathbb{C}^{\iota}$. For all points $x_1, \dots, x_N \in D$ and weights
$w_1, \dots, w_N \geq 0$ with $\sum_k w_k = 1$ there are at most $2m^2 + 1$ points
$y_j \in D$ and weights $v_j \geq 0$ with $\sum_j v_j = 1$ and
$\sum_j v_j \, a(y_j) a(y_j)^* = \sum_k w_k \, a(x_k) a(x_k)^*$.
\end{lemma}

\begin{proof}
\leanok
The Gram matrix lies in the convex hull of the matrices $a(y) a(y)^*$, so by
Carath\'eodory's theorem it is a convex combination of affinely independent ones, and an
affinely independent family in the real vector space $\mathbb{C}^{\iota \times \iota}$,
whose real dimension is $2m^2$, has at most $2m^2 + 1$ members.
\end{proof}

\begin{remark}[What the count is good for]
\label{rem:point-count}
\leanok
\uses{lem:few-design-points, thm:uniform-discretization, thm:uniform-discretization-compact}
The sharper count
$r + 1$, with $r$ the dimension of the span of the products $a_k \overline{a_l}$, is not
formalised, and nothing needs it.
\end{remark}

\begin{remark}[The identity decomposed over the design]
\label{rem:identity-decomposition}
\lean{Discretization.KieferWolfowitz.sum_weight_quadForm_inv_eq_card, Discretization.KieferWolfowitz.sum_weight_smul_quadForm_inv_eq_self}
\leanok
\uses{prop:almost-maximal-determinant, thm:uniform-discretization-compact}
For a design with points $x_1, \dots, x_n$, weights $w_1, \dots, w_n$ and invertible Gram
matrix $G$, and with $m = \#\iota$,
\[
  \sum_{k=1}^n w_k \, a(x_k)^* G^{-1} a(x_k) = m
  \qquad \text{and} \qquad
  \sum_{k=1}^n w_k \bigl( a(x_k)^* G^{-1} z \bigr) a(x_k) = z
\]
for every $z \in \mathbb{C}^{\iota}$, the first being $\Tr(G^{-1} G)$ and the second
$G G^{-1} z = z$ written out. With $c_k = w_k \, a(x_k)^* G^{-1} a(x_k)$ this is John's
decomposition of the identity over the design points, in the inner product
$\langle u, z \rangle = u^* G^{-1} z$ of the ellipsoid of $G$, and it is the condition
dual to the bound of \Cref{prop:almost-maximal-determinant}. Under
\Cref{thm:uniform-discretization-compact} the average forces
$a(x_k)^* G^{-1} a(x_k) = m$ at every design point, which makes the points contact
points. For an arbitrary convex body the theorem is proved in the companion project
\href{https://mario-ullrich.github.io/Lean-SNumbers/}{Lean-SNumbers}
(\href{https://mario-ullrich.github.io/Lean-SNumbers/blueprint.pdf}{PDF}).
\end{remark}
